### 1

Доказать, что $\left[ \vec{a}, \left[ \vec{b}, \vec{c} \right] \right] = \vec{b}(\vec{a}, \vec{c}) - \vec{c}(\vec{a}, \vec{b})$.

$$\begin{gather}
\left[ \vec{a}, \left[ \vec{b}, \vec{c} \right] \right] = \begin{vmatrix}
\vec{x}^{0} & \vec{y}^{0} & \vec{z}^{0} \\
a_{x} & a_{y} & a_{z} \\
\begin{vmatrix}
b_{y} & b_{z} \\
c_{y} & c_{z}
\end{vmatrix} & -\begin{vmatrix}
b_{x} & b_{z} \\
c_{x} & c_{z}
\end{vmatrix} & \begin{vmatrix}
b_{x} & b_{y} \\
c_{x} & c_{y}
\end{vmatrix}
\end{vmatrix} = \\
= \begin{vmatrix}
\vec{x}^{0} & \vec{y}^{0} & \vec{z}^{0} \\
a_{x} & a_{y} & a_{z} \\
b_{y}c_{z}-b_{z}c_{y} & b_{z}c_{x} - b_{x}c_{z} & b_{x}c_{y} - b_{y}c_{x}
\end{vmatrix} = \\
\end{gather}$$

$$\begin{gather}
= \vec{x}^{0}(a_{y}(b_{x}c_{y} - b_{y}c_{x}) - a_{z}(b_{z}c_{x} - b_{x}c_{z})) - \\
- \vec{y}^{0}(a_{x}(b_{x}c_{y} - b_{y}c_{x}) - a_{z}(b_{y}c_{z} - b_{z}c_{y})) + \\
+ \vec{z}^{0}(a_{x}(b_{z}c_{x} - b_{x}c_{z}) + a_{y}(b_{y}c_{z} - b_{z}c_{y})) = \\
\end{gather}$$

$$\begin{gather}
= \vec{x}^{0}b_{x}(a_{y}c_{y}+a_{z}c_{z}) - \vec{x}^{0}c_{x}(a_{y}b_{y} + a_{z}b_{z}) + \\
+ \vec{y}^{0}b_{y}(a_{x}c_{x}+a_{z}c_{z}) - \vec{y}^{0}c_{y}(a_{x}b_{x} + a_{z}b_{z}) + \\
+ \vec{z}^{0}b_{z}(a_{x}c_{x} + a_{y}c_{y}) - \vec{z}^{0}c_{z}(a_{x}b_{x} + a_{y}b_{y}) = 
\end{gather}$$
