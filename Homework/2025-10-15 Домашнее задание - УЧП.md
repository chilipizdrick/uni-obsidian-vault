$$\begin{gather}
u_{xx} + 8u_{yy} = 0, \ x \in (0,5), \ y \in(0,2), \\
u_{x}(0,y) = 10, \ u_{x}(2,y) = (1-y), \\
u_{y}(x,0) = 0, \ u(x,1) = 0
\end{gather}$$

$$\begin{gather}
u = X(x)Y(y) \\
X''Y + XY'' = 0 \\
\frac{X''}{X} + \frac{Y''}{Y} = 0 \\
\frac{X''}{X} = -\frac{Y''}{Y} = \lambda = \mathrm{const} \\
\lambda = -\mu^{2} \\
\begin{cases}
\frac{X''}{X} = -\mu^{2} \\
\frac{Y''}{Y} = \mu^{2}
\end{cases}
\end{gather}$$

$$\begin{gather}
\begin{cases}
X'' + \mu^{2}X = 0 \\
Y'' - \mu^{2}Y = 0
\end{cases}
\end{gather}$$
