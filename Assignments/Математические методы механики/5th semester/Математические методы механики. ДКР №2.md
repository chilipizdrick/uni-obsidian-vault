$$\begin{gather}
v_{M} = v_{r_{3}} = \omega_{3}\cdot r_{3} \\
v_{1} = v_{R_{2}} = v_{R_{3}} \\
\omega_{3} = \frac{v_{R_{3}}}{R_{3}} \implies v_{M} = \frac{v_{1}}{R_{3}} \cdot r_{3} \\
v_{1} = \frac{dx}{dt} \implies v_{M} = \frac{\mathrm{d}x}{\mathrm{d}t}\cdot \frac{r_{3}}{R_{3}} \\
\omega_{M} = \frac{\mathrm{d}x}{\mathrm{d}t}\cdot \frac{1}{R_{3}}
\end{gather}$$

$$\begin{gather}
v_{M} = 80t\cdot \frac{30}{45} = \frac{160}{3}t \ \frac{см}{с} \\
\end{gather}$$

$$\begin{gather}
\omega_{M} = \frac{80t}{45} = \frac{16}{9}t \ \frac{рад}{с} \\
a_{M}^{\tau} = \epsilon_{M} r_{M} = \frac{d\omega_{M}}{dt}\cdot r_{M} \\
a^{\tau}_{M} = \frac{d\omega_{M}}{dt}\cdot r_{3}
\end{gather}$$

$$\begin{gather}
a^{\tau}_{M} = \frac{16}{9}\cdot 30 = \frac{160}{3} \, \frac{см}{с^2}
\end{gather}$$

$$\begin{gather}
a^{n}_{M} = \omega_{M}^{2}r_{M} \\

\end{gather}$$

$$\begin{gather}
