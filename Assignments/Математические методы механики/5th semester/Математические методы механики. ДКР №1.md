$$\begin{gather}
x(t_{0}) = 2e^{0} = 2, \ y(t_{0}) = e^{2\cdot 0} - 1 = 0 \\
x(t_{1}) = 2e \approx 5.44, \ y(t_{1}) = e^{2}-1 \approx 6.39
\end{gather}$$

$$\begin{gather}
x = 2e^{t} \\
\frac{x}{2} = e^{t} \implies t = \ln \frac{x}{2} \\
y = e^{2 \ln x/2} - 1 \\
y = e^{\ln x^{2}/4} - 1 \\
y = \frac{x^{2}}{4} - 1 - \text{парабола} \\
\end{gather}$$

$$\begin{gather}
\dot{x} = 2e^{t} \\
\dot{y} = 2e^{2t}
\end{gather}$$

$$\begin{gather}
\vec{v}(t) = (2e^{t}, 2e^{2t}) \\
\vec{v}(t_{1}) = (2e,2e^{2}) \approx (5.44, 14.78) \\
\end{gather}$$

$$\begin{gather}
v(t) = 2e^{t}\sqrt{ 1 + e^{2t} } \\
v(t_{1}) = 2e\sqrt{ 1+e^{2} } \approx 15.75 \\
\end{gather}$$

$$\begin{gather}
\ddot{x} = 2e^{t} \\
\ddot{y} = 4e^{2t} \\
\end{gather}$$

$$\begin{gather}
\vec{a}(t) = (2e^{t}, 4e^{2t}) \\
\vec{a}(t_{1}) = (2e, 4e^{2}) \approx (5.44, 29.56) \\
\end{gather}$$

$$\begin{gather}
a(t) = 2e^{t}\sqrt{ 1 + 4e^{2t} } \\
a(t_{1}) = 2e\sqrt{ 1+4e^{2} } \approx 30.05
\end{gather}$$

$$\begin{gather}
\vec{a}_{\tau} = \frac{\mathrm{d}v}{\mathrm{d}t} \cdot \vec{\tau} \\
\frac{\mathrm{d}v}{\mathrm{d}t} = \frac{2e^{t}(2e^{2t}+1)}{\sqrt{ 1+e^{2t} }} \\
\vec{\tau} = \frac{1}{2e^{t}\sqrt{ 1+e^{2t} }}(2e^{t},2e^{2t}) = \frac{1}{\sqrt{ 1+e^{2t} }}(1,e^{t}) \\
\end{gather}$$

$$\begin{gather}
\vec{a}_{\tau}(t) = \frac{2e^{t}(2e^{2t}+1)}{\sqrt{ 1+e^{2t} }}\cdot \frac{1}{\sqrt{ 1+e^{2t} }}(1,e^{t}) = \frac{4e^{3t} + 2e^{t}}{1+e^{2t}}(1,e^{t}) \\
\vec{a}_{\tau}(t_{1}) = \frac{4e^{3} + 2e}{1+e^{2}}(1, e) \approx (10.23, 27.81) \\
\end{gather}$$

$$\begin{gather}
a_{\tau}(t) = \frac{4e^{3t} + 2e^{t}}{1+e^{2t}} \cdot \sqrt{ 1 + e^{2t} } =  \frac{4e^{3t} + 2e^{t}}{\sqrt{ 1+e^{2t} }}\\
a_{\tau}(t_{1}) = \frac{4e^{3}+2e}{\sqrt{ 1+e^{2} }} \approx 29.62
\end{gather}$$

$$\begin{gather}
R(t) = \frac{v(t)^{3}}{|\vec{v} \times \vec{a}|} \\
R(t) = \frac{(2e^{t}\sqrt{ 1 + e^{2t} })^{3}}{2e^{t}\cdot 4e^{2t} - 2e^{2t} \cdot 2e^{t}} = \frac{8e^{3t}(1+e^{2t})^{3/2}}{8e^{3t} - 4e^{3t}} = 2(1+e^{2t})^{3/2} \\
R(t_{1}) = 2(1+e^{2})^{3/2} \approx 48.6
\end{gather}$$
