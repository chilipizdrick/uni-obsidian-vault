$$\begin{gather}

\Phi(\phi) = C_{1}\cos \mu \phi + C_{2}\sin \mu \phi \\
\end{gather}$$

$$\begin{gather}

\end{gather}$$

При $n = 0$: $\sin 0 = 0, \ \Phi_{0} = 0$.

\Phi_{n}(\phi) = \sin 6n\phi - \text{ЛНЗ решения}\\

$$\begin{gather} u(r,\phi) = \sin(0)\cdot (\dots) + \sum_{n=1}^{\infty} \sin (6n\phi) \cdot (A_{n}r^{6n} + B_{n}r^{-6n}) \\ \end{gather}$$

### I.

### II.

$$\begin{gather}

$$\begin{gather}
c_{n} = \frac{\overbrace{\int_{0}^{\pi/2} (2-y)\sin 2ny \, dy}^{I_{1}}}{\underbrace{\int_{0}^{\pi/2} \sin ^{2} 2ny \, dy}_{I_{2}}} \\

$$\begin{gather}
I_{1} = \int_{0}^{\pi/2} (2-y)\sin 2ny \, dy = \\

$$\begin{gather}

-\frac{1}{n} [(-1)^{n} - 1] + \frac{\pi}{4n}\cdot (-1)^{n} - \left. \frac{1}{4n^{2}}\sin 2ny \right|_{0}^{\pi/2} = \\

$$\begin{gather}

$$\begin{gather}
I_{2} = \int_{0}^{\pi/2} \sin^{2} 2ny \, dy = \frac{1}{2} \int_{0}^{\pi/2} ( 1- \cos 4ny )  \, dy = \\

$$\begin{gather}
c_{n} = \frac{4}{\pi} \cdot \frac{(-1)^{n}(\pi - 4) + 4}{4n} = \frac{(-1)^{n}(\pi-4) + 4}{\pi n}
\end{gather}$$

$$\begin{gather}
2-y = \sum_{n=1}^{\infty} \frac{(-1)^{n}(\pi-4) + 4}{\pi n}\sin 2ny
\end{gather}$$

$$\begin{gather}
\frac{\partial^{2}}{\partial x^{2}}\left( \sum_{n=1}^{\infty}A_{n}(x)\sin 2ny \right) + \frac{\partial^{2}}{\partial y^{2}}\left( \sum_{n=1}^{\infty}A_{n}(x)\sin 2ny \right) = \\

$$\begin{gather}
A''_{n}(x) - 4n^{2}A_{n}(x) = \frac{(-1)^{n}(\pi-4) + 4}{\pi n} \\

$$\begin{gather}
-4n^{2}A_{чн_{n}}(x) = \frac{(-1)^{n}(\pi-4) + 4}{\pi n} \\
A_{чн_{n}} = -\frac{(-1)^{n}(\pi-4) + 4}{4\pi n^{3}} \\
A_{оо_{n}}(x) = B^{*}_{n}e^{2nx} + D^{*}_{n}e^{-2nx} \\
A_{оо_{n}}(x) = B_{n}\mathrm{ch} \, 2nx + D_{n}\mathrm{sh} \, 2nx \\

$$\begin{gather}
A_{n}(x) = B_{n}\mathrm{ch} \, 2nx + D_{n}\mathrm{sh} \, 2nx - \frac{(-1)^{n}(\pi-4)+4}{4\pi n^{3}} \\

$$\begin{gather}
A_{n}(0) = 0 \\
B_{n} - \frac{(-1)^{n}(\pi-4)+4}{4\pi n^{3}} = 0 \\
B_{n} = \frac{(-1)^{n}(\pi-4)+4}{4\pi n^{3}}

$$\begin{gather}
\begin{cases}

Введем

$$d_{n} = \begin{cases}

\frac{c_{n}}{4n^{2}}\mathrm{ch}\,4n + D_{n}\mathrm{sh}\, 4n - \frac{c_{n}}{4n^{2}} = d_{n} \\

$$\begin{gather}

$$\begin{gather}
u_{1} = \frac{1}{2\sqrt{ \pi t }}\int_{-\infty}^{\infty} e^{-\frac{x^{2} -2xy + y^{2} - 4ty^{2} - 4ty}{4t}} \, dy

$$\begin{gather}
u_{1} = \frac{1}{2\sqrt{ \pi t }}\int_{-\infty}^{\infty} e^{-\frac{1-4t}{4t}y^{2}-\frac{-x-2t}{2t}y-\frac{x^{2}}{4t}} \, dy

Используем шаблонный интеграл:

$$\begin{gather}
I = \int_{-\infty}^{\infty} e^{-ay^{2}-by-c} \, dy = e^{\frac{b^{2}-4ac}{4a}}\sqrt{ \frac{\pi}{a} }\\
\end{gather}$$

В нашем случае:

$$\begin{gather}
\begin{cases}
a = \frac{1-4t}{4t} \\
b = \frac{-x-2t}{2t} \\
c = \frac{x^{2}}{4t}
\end{cases} \\
\end{gather}$$

$$\begin{gather}
u_{1} =  \frac{1}{2\sqrt{ \pi t }}\cdot \sqrt{ \frac{4\pi t}{1-4t} }\cdot e^{\frac{\frac{(x+2t)^{2}}{4t^{2}} - 4\cdot \frac{1-4t}{4t} \cdot \frac{x^{2}}{4t}}{4\cdot \frac{1-4t}{4t}}} = \\
= \frac{1}{\sqrt{ 1-4t }} e^{\frac{x^{2}+x+t}{1-4t}}

$$\begin{gather}
u_{2} = \frac{1}{2\sqrt{ \pi }}\int_{0}^{t} \frac{d\tau}{\sqrt{ t-\tau }} \int_{-\infty}^{+\infty} e^{-\tau}\sin y \cdot e^{-\frac{(x-y)^{2}}{4(t-\tau)}}\,dy \\

$$\begin{gather}
I_{1} = \int_{-\infty}^{+\infty} \sin y \cdot e^{-\frac{y^{2}-2xy}{4(t-\tau)}}\,dy = \\
= \frac{1}{2i}\int_{-\infty}^{+\infty} \left[e^{iy-\frac{y^{2}-2xy}{4(t-\tau)}} - К.С.\right] \, dy = \\
= \frac{1}{2i}\left[\underbrace{\int_{-\infty}^{+\infty} e^{iy-\frac{y^{2}-2xy}{4(t-\tau)}} \, dy}_{I_{1}'} - К.С.И. \right]
\end{gather}$$

$$I = \int_{-\infty}^{\infty} e^{-ay^{2}-by-c} \, dy = e^{\frac{b^{2}-4ac}{4a}}\sqrt{ \frac{\pi}{a} }$$

$$\begin{gather}
I_{1}' = \int_{-\infty}^{+\infty} e^{iy-\frac{y^{2}-2xy}{4(t-\tau)}} \, dy = \int_{-\infty}^{+\infty} e^{-\frac{1}{4(t-\tau)}y^{2} -\left( -i-\frac{2x}{4(t-\tau)} \right)y} \, dy = \\
