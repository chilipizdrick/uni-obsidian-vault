### Вариант 5

$$\begin{cases}
u_{t} = u_{xx} + f(t,x), \ x \in \mathbb{R}, \ t > 0, \\
u(0,x) = \phi(x), \ x \in \mathbb{R}.
\end{cases}$$

$$\begin{align}
&f(t,x) = e^{-t}\sin x \\
&\phi(x) = e^{-x^{2}-x}
\end{align}$$

$$\begin{align}
&u = u_{1} + u_{2} \\
&u_{1} = \frac{1}{2\sqrt{ \pi t }} \int_{-\infty}^{\infty} \phi(y) e^{-\frac{(x-y)^{2}}{4t}}\, dy \\
&u_{2} = \int_{0}^{t} \frac{d\tau}{2\sqrt{ \pi(t-\tau) }} \int_{-\infty}^{+\infty} f(\tau,y) \cdot e^{-\frac{(x-y)^{2}}{4(t-\tau)}} \, dy
\end{align}$$

$$\begin{gather}
u_{1} = \frac{1}{2\sqrt{ \pi t }}\int_{-\infty}^{\infty} e^{-y^{2}-y}\cdot e^{-\frac{(x-y)^{2}}{4t}} \, dy \\
\end{gather}$$

$$\begin{gather}
u_{1} = \frac{1}{2\sqrt{ \pi t }}\int_{-\infty}^{\infty} e^{-\frac{x^{2} -2xy + y^{2} - 4ty^{2} - 4ty}{4t}} \, dy
\end{gather}$$

$$\begin{gather}
u_{1} = \frac{1}{2\sqrt{ \pi t }}\int_{-\infty}^{\infty} e^{-\frac{1-4t}{4t}y^{2}-\frac{-2x-4t}{4t}y-\frac{x^{2}}{4t}} \, dy
\end{gather}$$

Ипользуем шаблонный интеграл:

$$\begin{gather}
I(y) = \int_{-\infty}^{\infty} e^{-ay^{2}-by-c} \, dy = e^{\frac{b^{2}-4ac}{4a}}\sqrt{ \frac{\pi}{a} }\\
\end{gather}$$

В нашем случае:

$$\begin{gather}
\begin{cases}
a = \frac{1-4t}{4t} \\
b = -\frac{2x+4t}{4t} \\
c = \frac{x^{2}}{4t}
\end{cases} \\
\end{gather}$$

$$\begin{gather}
u_{1} =  \frac{1}{\sqrt{ 1-4t }} e^{\frac{\frac{(x+2t)^{2}}{4t^{2}} - 4\cdot \frac{1-4t}{4t} \cdot \frac{x^{2}}{4t}}{4\cdot \frac{1-4t}{4t}}} = \\
= \frac{1}{\sqrt{ 1-4t }} e^{\frac{(x+2t)^{2} - (1-4t)x^{2}}{4t(1-4t)}} = \\
\end{gather}$$

$$\begin{gather}
= \frac{1}{\sqrt{ 1-4t }} e^{\frac{x^{2} + x + 4t}{1-4t}} \\
\end{gather}$$

$$\begin{gather}
u_{2} = \int_{0}^{t} \frac{d\tau}{2\sqrt{ \pi(t-\tau) }} \int_{-\infty}^{+\infty} e^{-\tau}\sin y \cdot e^{-\frac{(x-y)^{2}}{4(t-\tau)}}\,dy \\
u_{2} = \int_{0}^{t} \frac{e^{-\tau -\frac{x^{2}}{4(t-\tau)}} \, d\tau}{2\sqrt{ \pi(t-\tau) }} \underbrace{\int \sin y \cdot e^{-\frac{y^{2}-2xy}{4(t-\tau)}}\,dy}_{I_{1}}  \\
\end{gather}$$

$$\begin{gather}
I_{1} = \int_{-\infty}^{+\infty} \sin y \cdot e^{-\frac{y^{2}-2xy}{4(t-\tau)}}\,dy = \\
= \frac{1}{2i}\int_{-\infty}^{+\infty} \left[e^{iy-\frac{y^{2}-2xy}{4(t-\tau)}} - К.С.\right] \, dy = \\
= \frac{1}{2i}\left[\underbrace{\int_{-\infty}^{+\infty} e^{iy-\frac{y^{2}-2xy}{4(t-\tau)}} \, dy}_{I_{1}'} - К.С.И. \right]
\end{gather}$$

$$I(y) = \int_{-\infty}^{\infty} e^{-ay^{2}-by-c} \, dy = e^{\frac{b^{2}-4ac}{4a}}\sqrt{ \frac{\pi}{a} }$$

$$\begin{gather}
I_{1}' = \int_{-\infty}^{+\infty} e^{iy-\frac{y^{2}-2xy}{4(t-\tau)}} \, dy = \int_{-\infty}^{+\infty} e^{-\frac{1}{4(t-\tau)}y^{2} -\left( -i-\frac{2x}{4(t-\tau)} \right)y} \, dy = \\
= \sqrt{ 4\pi(t-\tau) }\cdot e^{\left( i+\frac{2x}{4(t-\tau)} \right)^{2}(t-\tau)} = \\
\end{gather}$$

$$\begin{gather}
= \sqrt{ 4\pi(t-\tau) }\cdot e^{\left( -1 + \frac{ix}{t-\tau} + \frac{4x^{2}}{16(t-\tau)^{2}} \right)(t-\tau)} = \\
= \sqrt{ 4\pi(t-\tau) }\cdot e^{-(t-\tau) + ix + \frac{x^{2}}{4(t-\tau)}} = \\
= 2\sqrt{ \pi(t-\tau) }\cdot e^{\frac{x^{2}}{4(t-\tau)}-(t-\tau) }e^{ix}
\end{gather}$$

$$\begin{gather}
I_{1} = 2\sqrt{ \pi(t-\tau) } \cdot e^{\frac{x^{2}}{4(t-\tau)}-(t-\tau)} \frac{e^{ix} - e^{-ix}}{2i} = \\
= 2\sqrt{ \pi(t-\tau) } \cdot e^{\frac{x^{2}}{4(t-\tau)}-(t-\tau)} \sin x
\end{gather}$$

$$\begin{gather}
u_{2} = \int_{0}^{t} \frac{e^{-\tau -\frac{x^{2}}{4(t-\tau)}}}{2\sqrt{ \pi(t-\tau) }} \left(2\sqrt{ \pi(t-\tau) } \cdot e^{\frac{x^{2}}{4(t-\tau)}-(t-\tau)} \sin x\right) \, d\tau = \\
= e^{-t}\sin x\int_{0}^{t} d\tau = te^{-t}\sin x \\
\end{gather}$$

$$\begin{gather}
u(t,x) = \frac{e^{\frac{x^{2} + x + 4t}{1-4t}}}{\sqrt{ 1-4t }} + te^{-t}\sin x
\end{gather}$$
