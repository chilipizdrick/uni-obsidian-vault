### I.

$$\begin{gather}
u_{xx} + u_{yy} = 0 \\
u = X(x)Y(y) \\
X''Y + XY'' = 0 \\
\frac{X''}{X} + \frac{Y''}{Y} = 0 \\
\frac{X''}{X} = -\frac{Y''}{Y} = \mathrm{const} = \lambda = \mu^{2} \\
\begin{cases}
X'' -\mu^{2}X = 0 \\
Y'' +\mu^{2}Y = 0 \\
\end{cases}
\end{gather}$$

$$\begin{gather}
Y'' + \mu^{2}Y = 0 \\
Y = C_{1}\cos \mu y + C_{2}\sin \mu y \\
Y(0) = C_{1} \cos 0 + C_{2} \sin 0 = 0 \implies C_{1} = 0 \\
Y\left( \frac{\pi}{2} \right) = C_{2} \sin \frac{\mu \pi}{2} = 0 \implies \mu = 2n, \ n = 1,2,3,\dots \\
Y_{n}(y) = C_{2,n}\sin 2 ny, \ n = 1,2,3,\dots \\
Y_{n}(y) = \sin 2 ny, \ n = 1,2,3,\dots - \text{ЛНЗ решения}
\end{gather}$$

### II.

$$\begin{gather}
u(x,y) = \sum_{n=1}^{\infty} A_{k}(x)\sin 2ny \\
\end{gather}$$

$$\begin{gather}
2-y = \sum_{n=1}^{\infty} c_{n}\sin 2ny, \  c_{n} = \frac{((2-y), \sin 2ny)}{||\sin 2ny||^{2}} \\
\end{gather}$$

$$\begin{gather}
c_{n} = \frac{\overbrace{\int_{0}^{\pi/2} (2-y)\sin 2ny \, dy}^{I_{1}}}{\underbrace{\int_{0}^{\pi/2} \sin ^{2} 2ny \, dy}_{I_{2}}} \\
\end{gather}$$

$$\begin{gather}
I_{1} = \int_{0}^{\pi/2} (2-y)\sin 2ny \, dy = \\
= 2\int_{0}^{\pi/2} \sin 2ny \, dy - \int_{0}^{\pi/2} y \sin 2ny \, dy = \\
\end{gather}$$

$$\begin{gather}
= \left. -\frac{1}{n} \cos 2ny \right|_{0}^{\pi/2} - \left[ \left. -y\frac{\cos 2ny}{2n} \right|_{0}^{\pi/2} + \frac{1}{2n}\int_{0}^{\pi/2} \cos 2ny \, dy \right] = \\
-\frac{1}{n} [(-1)^{n} - 1] + \frac{\pi}{4n}\cdot (-1)^{n} - \left. \frac{1}{4n^{2}}\sin 2ny \right|_{0}^{\pi/2} = \\
\end{gather}$$

$$\begin{gather}
= \frac{(-1)^{n}(\pi - 4) + 4}{4n}
\end{gather}$$

$$\begin{gather}
I_{2} = \int_{0}^{\pi/2} \sin^{2} 2ny \, dy = \frac{1}{2} \int_{0}^{\pi/2} ( 1- \cos 4ny )  \, dy = \\
= \frac{1}{2} \left[ \frac{\pi}{2} - \frac{1}{4n}\sin 4ny|_{0}^{\pi/2} \right] = \frac{\pi}{4}
\end{gather}$$

$$\begin{gather}
c_{n} = \frac{4}{\pi} \cdot \frac{(-1)^{n}(\pi - 4) + 4}{4n} = \frac{(-1)^{n}(\pi-4) + 4}{\pi n}
\end{gather}$$

$$\begin{gather}
2-y = \sum_{n=1}^{\infty} \frac{(-1)^{n}(\pi-4) + 4}{\pi n}\sin 2ny
\end{gather}$$

$$\begin{gather}
\frac{\partial^{2}}{\partial x^{2}}\left( \sum_{n=1}^{\infty}A_{n}(x)\sin 2ny \right) + \frac{\partial^{2}}{\partial y^{2}}\left( \sum_{n=1}^{\infty}A_{n}(x)\sin 2ny \right) = \\
= \sum_{n=1}^{\infty} \frac{(-1)^{n}(\pi-4) + 4}{\pi n}\sin 2ny
\end{gather}$$

$$\begin{gather}
A''_{n}(x) - 4n^{2}A_{n}(x) = \frac{(-1)^{n}(\pi-4) + 4}{\pi n} \\
\lambda^{2} - 4n^{2} = 0 \implies \lambda_{1,2} = \pm 2n \\
\end{gather}$$

$$\begin{gather}
-4n^{2}A_{чн_{n}}(x) = \frac{(-1)^{n}(\pi-4) + 4}{\pi n} \\
A_{чн_{n}} = -\frac{(-1)^{n}(\pi-4) + 4}{4\pi n^{3}} \\
A_{оо_{n}}(x) = B^{*}_{n}e^{2nx} + D^{*}_{n}e^{-2nx} \\
A_{оо_{n}}(x) = B_{n}\mathrm{ch} \, 2nx + D_{n}\mathrm{sh} \, 2nx \\
\end{gather}$$

$$\begin{gather}
A_{n}(x) = B_{n}\mathrm{ch} \, 2nx + D_{n}\mathrm{sh} \, 2nx - \frac{(-1)^{n}(\pi-4)+4}{4\pi n^{3}} \\
\end{gather}$$

$$\begin{gather}
A_{n}(0) = 0 \\
B_{n} - \frac{(-1)^{n}(\pi-4)+4}{4\pi n^{3}} = 0 \\
B_{n} = \frac{(-1)^{n}(\pi-4)+4}{4\pi n^{3}}
\end{gather}$$

$$\begin{gather}
\begin{cases}
A_{2}(2) = 1 \\
A_{n}(2) = 0, \ n \neq 2
\end{cases} \\
\end{gather}$$

Введем

$$d_{n} = \begin{cases}
1, n = 2 \\
0, n \neq 2
\end{cases}$$

Тогда 

$$\begin{gather}
\frac{c_{n}}{4n^{2}}\mathrm{ch}\,4n + D_{n}\mathrm{sh}\, 4n - \frac{c_{n}}{4n^{2}} = d_{n} \\
\end{gather}$$

$$\begin{gather}
D_{n} = \frac{d_{n} + \frac{c_{n}}{4n^{2}}(1-\mathrm{ch}\, 4n)}{\mathrm{sh}\, 4k}, \ \text{где} \  c_{n} = \frac{(-1)^{n}(\pi-4) + 4}{\pi n}
\end{gather}$$

Наконец

$$\begin{gather}
A_{n}(x) = \frac{c_{n}}{4n^{2}}(\mathrm{ch} \, 2nx - 1) + \frac{d_{n} + \frac{c_{n}}{4n^{2}}(1-\mathrm{ch}\, 4n)}{\mathrm{sh}\, 4k}\mathrm{sh} \, 2nx \\
\end{gather}$$

Тогда 

$$u(x,y) = \sum_{n=1}^{\infty} \left[ \frac{c_{n}}{4n^{2}}(\mathrm{ch} \, 2nx - 1) + \frac{d_{n} + \frac{c_{n}}{4n^{2}}(1-\mathrm{ch}\, 4n)}{\mathrm{sh}\, 4k}\mathrm{sh} \, 2nx \right] \sin 2ny,$$

где $c_{n} = \frac{(-1)^{n}(\pi-4) + 4}{\pi n}$, $d_{n} = \begin{cases}1, n = 2 \\ 0, n \neq 2 \end{cases}$.
