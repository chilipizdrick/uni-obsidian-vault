### Вариант 5

$$\begin{gather}
\begin{cases}
\frac{1}{a^{2}}u_{tt} = u_{xx} + u_{yy} + x-t, \ x^{2} + y^{2} < r_{0}^{2}, \ t > 0 \\
u|_{x^{2}+y^{2} = r_{0}^{2}} = 0, \\
u|_{t=0} = J_{0}\left( \frac{\mu_{5}^{(0)}r}{r_{0}} \right), \ u_{t}|_{t=0} = 0
\end{cases}
\end{gather}$$

### 1. Переход к полярным координатам

$$\begin{gather}
\begin{cases}
x = r\cos \phi \\
y = r\sin \phi
\end{cases} \\
\Delta = \frac{1}{r} \frac{\partial }{\partial r}\left( r \frac{\partial}{\partial r} \right) + \frac{1}{r^{2}} \frac{\partial^{2}}{\partial \phi^{2}}
\end{gather}$$

$$\begin{cases}
\frac{1}{a^{2}}u_{tt} - \frac{1}{r}(ru_{r})_{r} - \frac{1}{r^{2}}u_{\phi \phi} = r\cos \phi - t, \ r < r_{0}, \ t > 0 \\
u|_{r=r_{0}} = 0 \\
u|_{t=0} = J_{0}\left( \frac{\mu_{5}^{(0)}r}{r_{0}} \right), \ u_{t}|_{t=0} = 0
\end{cases}$$

### 2. Разбиение на подзадачи

$$\begin{gather}
u = V + W
\end{gather}$$

$$\begin{cases}
\frac{1}{a^{2}}V_{tt} - \frac{1}{r}(rV_{r})_{r} \frac{1}{r^{2}}V_{\phi \phi} = 0 \\
V|_{r=r_{0}} = 0, \ \left( V_{\phi+2\pi} = V|_{\phi} \right), \\
V|_{t=0} = J_{0}\left( \frac{\mu_{5}^{(0)}r}{r_{0}} \right), \ V_{t}|_{t=0} = 0
\end{cases}$$

$$\begin{cases}
\frac{1}{a^{2}}W_{tt} - \frac{1}{r}(rW_{r})_{r} \frac{1}{r^{2}}W_{\phi \phi} = r\cos \phi - t, \\
W|_{r=r_{0}} = 0, \left( W_{\phi+2\pi} = W|_{\phi} \right), \\
W|_{t=0} = 0, \ W_{t}|_{t=0} = 0
\end{cases}$$

### 3. Решение для $V$

$$\begin{gather}
V = R(r)\Phi(\phi)T(t) \\
\frac{1}{a^{2}}R\Phi T'' - \frac{1}{r}(rR')'\Phi T - \frac{1}{r^{2}}R\Phi''T = 0 \\
\end{gather}$$

$$\begin{gather}
\frac{1}{a^{2}}\cdot \frac{T''}{T} - \frac{1}{rR}(rR')' - \frac{\Phi''}{r^{2}\Phi} = 0
\end{gather}$$

$$\begin{gather}
\frac{\Phi''}{\Phi} = r^{2} \cdot \frac{1}{a^{2}} \frac{T''}{T} - \frac{r}{R} (rR')' = -\xi^{2} = \mathrm{const} \\
\Phi'' + \xi^{2}\Phi = 0
\end{gather}$$

$$\begin{gather}
r^{2}\cdot \frac{1}{a^{2}} \frac{T''}{T} - \frac{r}{R}(rR')' = -\xi^{2} \\
\frac{1}{a^{2}}\cdot \frac{T''}{T} = \frac{1}{rR}(rR')' - \frac{\xi^{2}}{r^{2}} = -\eta^{2} = \mathrm{const} \\
\end{gather}$$

$$\begin{cases}
\Phi'' + \xi^{2}\Phi = 0 \\
r(rR')' + (\eta^{2}r^{2} - \xi^{2})R = 0 \\
T'' + \eta^{2}a^{2}T = 0
\end{cases}$$

$$\begin{gather}
\begin{cases}
\Phi'' + \xi^{2}\Phi = 0 \\
\Phi(\phi + 2\pi) = \Phi(\phi) \\
\end{cases} \\
\end{gather}$$

Используем известное решение:

$$\begin{gather}
\Phi_{n} = e^{in\phi}, \\
\xi_{n} = n, \ n \in \mathbb{Z}
\end{gather}$$

$$\begin{gather}
r^{2}R'' + rR' + (\eta^{2}r^{2} - n^{2})R = 0 \\
\end{gather}$$

Используем замену $x = \eta r$:

$$\begin{gather}
x^{2}R'' + xR' + (x^{2} - n^{2})R = 0 \\
\end{gather}$$

Тогда 

$$\begin{gather}
R = C_{1}\cdot J_{n}(x) + \cancel{ {C_{2}Y_{n}(x)} }
\end{gather}$$

В таком разбиении функция Бесселя в нуле ограничена, а фукнция Неймана - нет, потому мы eё выбрасываем.

$$\begin{gather}
R(r) = C_{1}J_{n}(\eta r) \\
R(r_{0}) = 0, \\
R(r_{0}) = C_{1}J_{n}(\eta r_{0}) = 0, \\
\eta r_{0} = \mu^{(n)}_{m} \\
\end{gather}$$

$$\begin{gather}
R_{n,m}(r) = J_{n}\left( \frac{\mu_{m}^{(n)}r}{r_{0}} \right), \ \eta_{m} = \frac{\mu_{m}^{(n)}}{r_{0}}, \ m = 1,2,3,\dots
\end{gather}$$

$$\begin{gather}
T''_{n,m} + \left( \frac{\mu_{m}^{(n)}}{r_{0}}\cdot a  \right)^{2} T_{n,m} = 0
\end{gather}$$

$$\begin{gather}
T_{n,m} = A_{n,m}\cos\left( \frac{\mu_{m}^{(n)}a}{r_{0}}\cdot t \right) + B_{n,m}\sin\left( \frac{\mu_{m}^{(n)}a}{r_{0}} \cdot t \right) \\
\end{gather}$$

$$\begin{gather}
V = \sum_{n=-\infty}^{+\infty} \sum_{m=1}^{+\infty} \left( A_{n,m}\cos\left( \frac{\mu_{m}^{(n)}a}{r_{0}}t \right) + B_{n,m}\sin\left( \frac{\mu_{m}^{(n)}a}{r_{0}}t \right) \right) J_{n}\left( \frac{\mu_{m}^{(n)}r}{r_{0}} \right)e^{in\phi}
\end{gather}$$

$$\begin{gather}
V|_{t=0} = \sum_{n,m}A_{n,m}J_{n}\left( \frac{\mu_{m}^{(n)}r}{r_{0}} \right)e^{in\phi} = J_{0}\left( \frac{\mu_{5}^{(0)}r}{r_{0}} \right) e^{i\cdot 0 \cdot \phi} \\
\begin{cases}
A_{n,m} = 0, \ n \neq 0 \vee m \neq 5 \\
A_{0,5} = 1
\end{cases}
\end{gather}$$

$$\begin{gather}
V_{t}|_{t=0} = \sum_{n,m} B_{n,m}\cdot \frac{\mu_{m}^{(n)}a}{r_{0}}J_{n}\left( \frac{\mu_{m}^{(n)}r}{r_{0}} \right)e^{in\phi} = 0, \ \implies B_{n,m} = 0
\end{gather}$$

Откуда

$$\begin{gather}
V = \cos\left( \frac{\mu_{5}^{(0)}a}{r_{0}}t \right)J_{0}\left( \frac{\mu_{5}^{(0)}r}{r_{0}} \right)
\end{gather}$$

### 4. Решение для $W$

$$\begin{gather}
W = \sum_{n=-\infty}^{+\infty}\sum_{m=1}^{+\infty} w_{n,m}(t) J_{n}\left( \frac{\mu_{m}^{(n)}r}{r_{0}} \right)e^{in\phi}
\end{gather}$$

$$\begin{gather}
W_{tt} = \sum_{n,m} w_{n,m}'R_{n,m}(r)e^{in\phi} \\
\frac{1}{r}(rW_{r})_{r} = \frac{1}{r}\sum_{n,m}w_{n,m}(rR_{n,m}')' e^{in\phi} = \dots
\end{gather}$$

В силу того, что $r(rR')' + (r^{2}\eta^{2}-\xi^{2})R = 0$:

$$\begin{gather}
\dots = \frac{1}{r^{2}} \sum_{n,m} w_{n,m}\left( n^{2} - r^{2}\left(\frac{\mu_{m}^{(n)}}{r_{0}}\right)^{2} \right)R_{n,m}e^{in\phi} \\
\end{gather}$$

$$\begin{gather}
\frac{1}{r^{2}}W_{\phi \phi} = \frac{1}{r^{2}}\sum_{n,m}w_{n,m}R_{n,m}(-n^{2})e^{in\phi}
\end{gather}$$

$$\begin{gather}
\sum_{n,m} \left( \frac{1}{a^{2}}w''_{n,m} - \frac{1}{r^{2}}\left( n^{2} - \left( \frac{\mu_{m}^{(n)}}{r_{0}} \right)^{2} \right)w_{n,m} + \frac{n^{2}}{r^{2}}w_{n,m} \right)J_{n}\left( \frac{\mu_{m}^{(n)}r}{r_{0}} \right)e^{in\phi} \\

\end{gather}$$

$$\begin{gather}
\sum_{n,m} \underbrace{\left( \frac{1}{a^{2}}w''_{n,m} - \left( \frac{\mu_{m}^{(n)}}{r_{0}} \right)^{2}w_{n,m}  \right)}_{K_{n,m}(t)}J_{n}\left( \frac{\mu_{m}^{(n)}r}{r_{0}} \right)e^{in\phi} = r\cos\phi - t
\end{gather}$$

$$\begin{gather}
r\cos \phi - t = \frac{r}{2} e^{i \cdot 1 \cdot \phi} + \frac{r}{2} \cdot e^{i \cdot (-1) \cdot \phi} - t \cdot e^{i\cdot 0 \cdot \phi}
\end{gather}$$

$$\begin{gather}
\sum_{m} = K_{0,m}J_{0}\left( \frac{\mu_{m}^{(0)}r}{r_{0}} \right) = -t \\
\sum_{m} = K_{1,m}J_{1}\left( \frac{\mu_{m}^{(1)}r}{r_{0}} \right) = \frac{r}{2} \\
\sum_{m} = K_{-1,m}J_{-1}\left( \frac{\mu_{m}^{(-1)}r}{r_{0}} \right) = \frac{r}{2} \\
\end{gather}$$

$$\begin{gather}
K_{-n,m} = K_{n,m}^{*} \\
K_{n,m} = 0, \ n \neq 0, \pm 1
\end{gather}$$

$$\begin{gather}
K_{0,m} = \frac{\left( -t, \ J_{0}\left( \frac{\mu_{m}^{(0)}r}{r_{0}} \right) \right)}{\left|\left|J_{0}\left( \frac{\mu_{m}^{(0)}r}{r_{0}} \right)\right|\right|^{2}} = \frac{\int_{0}^{r_{0}} r\cdot (-t) \cdot J_{0}\left( \frac{\mu_{m}^{(0)}r}{r_{0}} \right) \, dr }{\int_{0}^{r_{0}} r\cdot J_{0}^{2}\left( \frac{\mu_{m}^{(0)}r}{r_{0}} \right) \, dr } = \alpha_{0,m}t
\end{gather}$$

Аналогично: 

$$\begin{gather}
K_{1,m} = \frac{\left( \frac{r}{2}, \ J_{1}\left( \frac{\mu_{m}^{(1)}r}{r_{0}} \right) \right)}{\left|\left|J_{1}\left( \frac{\mu_{m}^{(1)}r}{r_{0}} \right)\right|\right|^{2}} = \frac{\int_{0}^{r_{0}} r\cdot \frac{r}{2} \cdot J_{1}\left( \frac{\mu_{m}^{(1)}r}{r_{0}} \right) \, dr }{\int_{0}^{r_{0}} r\cdot J_{1}^{2}\left( \frac{\mu_{m}^{(1)}r}{r_{0}} \right) \, dr } = \alpha_{1,m}
\end{gather}$$

$$\begin{gather}
K_{-1,m} = \frac{\left( \frac{r}{2}, \ J_{-1}\left( \frac{\mu_{m}^{(-1)}r}{r_{0}} \right) \right)}{\left|\left|J_{-1}\left( \frac{\mu_{m}^{(-1)}r}{r_{0}} \right)\right|\right|^{2}} = \frac{\int_{0}^{r_{0}} r\cdot \frac{r}{2} \cdot J_{-1}\left( \frac{\mu_{m}^{(-1)}r}{r_{0}} \right) \, dr }{\int_{0}^{r_{0}} r\cdot J_{-1}^{2}\left( \frac{\mu_{m}^{(-1)}r}{r_{0}} \right) \, dr } = \alpha_{-1,m}
\end{gather}$$

Получили функции $V$ и $W$, $u = V + W$.

$$\begin{gather}
u = \cos\left( \frac{\mu_{5}^{(0)}a}{r_{0}}t \right)J_{0}\left( \frac{\mu_{5}^{(0)}r}{r_{0}} \right) + \\
+ \sum_{m=1}^{+\infty}\left(\alpha_{-1,m}J_{-1}\left( \frac{\mu_{m}^{(-1)}r}{r_{0}} \right)e^{-i\phi} + \alpha_{0,m}\cdot t\cdot J_{0}\left( \frac{\mu_{m}^{(0)}r}{r_{0}} \right) + \alpha_{1,m}\left( \frac{\mu_{m}^{(1)}r}{r_{0}} \right)e^{i\phi} \right)
\end{gather}$$
