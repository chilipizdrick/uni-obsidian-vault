$$\begin{cases}

\end{cases}$$

### Решение

$$\begin{gather}
u = H + V
\end{gather}$$

$$\begin{gather}
H = a(t) + x\cdot b(t) \\
H(0,t) = 0, \ H(\pi,t) = e^{t} - 1
\end{gather}$$

$$\begin{gather}
\begin{cases}
a(t) + 0\cdot b(t) = 0 \\
a(t) + \pi \cdot b(t) = e^{t} - 1
\end{cases} \implies \begin{cases}
a(t) = 0 \\
b(t) = \frac{e^{t}-1}{\pi}
\end{cases} \\
H(x,t) = x\cdot \frac{e^{t}-1}{\pi}
\end{gather}$$

$$\begin{cases}
H_{tt} + V_{tt} - H_{xx} - V_{xx} = 2x \\
H(0,t) + V(0,t) = 0, \ H(\pi,t) + V(\pi,t) = e^{t}-1 \\
H(x,0) + V(x,0) = 0, H_{t}(x,0) + V_{t}(x,0) = \frac{1-\cos x}{2}
\end{cases}$$

$$\begin{cases}
V_{tt} - V_{xx} = 2x - \frac{x(e^{t}-1)}{\pi} \\
V(0,t) = 0, \ V(\pi,t) = 0 \\
V(x,0) = 0, \ V_{t}(x,0) = \frac{1-\cos x}{2}
\end{cases}$$

$$V = W + P$$

$$\begin{cases}
W_{tt} - W_{xx} = 0 \\
W(0,t) = 0, \ W(\pi,t) = 0 \\
W(x,0) = 0, \ W_{t}(x,0) = \frac{1 - \cos x}{2}
\end{cases}$$

$$\begin{cases}
P_{tt} - P_{xx} = 2x - \frac{x(e^{t}-1)}{\pi} \\
P(0,t) = 0, \ P(\pi,t) = 0 \\
P(x,0) = 0, \ P_{t}(x,0) = 0
\end{cases}$$

$$\begin{gather}
W(x,t) = X(x)T(t) \\
XT'' - X''T = 0 \\
\frac{T''}{T} - \frac{X''}{X} = 0 \\
\frac{T''}{T} = \frac{X''}{X} = -\mu^{2} = \mathrm{const} \\
\end{gather}$$

$$\begin{gather}
\begin{cases}
T'' + \mu^{2}T = 0 \\
X'' + \mu^{2}X = 0
\end{cases} \\
X'' + \mu^{2}X = 0 \\
X(x) = C_{1}\cos(\mu x) + C_{2}\sin(\mu x) \\
X(0) = C_{1} = 0 \implies C_{1} = 0 \\
X(\pi) = C_{2}\sin(\mu \pi) = 0 \implies \mu \pi = \pi n \implies \mu = n \\
X_{n}(x) = \sin(nx), \ n = 1,2,3,\dots
\end{gather}$$

$$\begin{gather}
W(x,t) = \sum_{n=1}^{+\infty} [A_{n}\cos(nt) + B_{n}\sin(nt)]\sin(nx) \\
W_{t}(x,t) = \sum_{n=1}^{\infty}[-nA_{n}\sin(nt) + nB_{n}\cos(nt)]\sin(nx)
\end{gather}$$

$$\begin{gather}

W_{t}(x,0) = \sum_{n=1}^{\infty} nB_{n}\sin(nx) = \frac{1-\cos x}{2}
\end{gather}$$

$$\begin{gather}

$$\begin{gather}

\end{gather}$$

$$\begin{gather}
= -\frac{(-1)^{n} - 1}{2n} - \\
- \frac{1}{4}\int_{0}^{\pi} \sin((n+1)x) \, dx  - \frac{1}{4}\int_{0}^{\pi} \sin((n-1)x) \, dx = \\
= \frac{1-(-1)^{n}}{2n} + \\
+ \frac{1}{4(n+1)}\cos((n+1)x)|_{0}^{\pi} + \frac{1}{4(n-1)}\cos((n-1)x)|_{0}^{\pi} =
\end{gather}$$

$$\begin{gather}

$$I_{2} = \int_{0}^{\pi} \sin ^{2}(nx) \, dx = \frac{\pi}{2}$$

$$\begin{gather}
W(x,t) = \sum_{n=1}^{+\infty} \frac{(-1)^{n}(\pi n^{2} + 1)}{2n^{2}(1-n^{2})}\sin(nt) \sin(nx)
\end{gather}$$

$$\begin{cases}
P_{tt} - P_{xx} = 2x - \frac{x(e^{t}-1)}{\pi} \\
P(0,t) = 0, \ P(\pi,t) = 0 \\
P(x,0) = 0, \ P_{t}(x,0) = 0

$$\begin{gather}

$$\begin{gather}

$$\begin{gather}

P^{чн}_{n} = C_{n} + D_{n}e^{t} \\
\end{gather}$$

$$\begin{gather}

\begin{cases}

(1+n^{2})D_{n} = -\frac{(-1)^{n+1}}{n}

D_{n} = \frac{-(-1)^{n+1}}{n(1+n^{2})}

\end{gather}$$

$$\begin{gather}
P_{n} = A_{n}\cos(nt) + B_{n}\sin(nt) -\frac{5(-1)^{n}}{n^{3}} + \frac{(-1)^{n}}{n(1+n^{2})}e^{t} \\
P_{n} = A_{n}\cos(nt) + B_{n}\sin(nt) - \alpha_{n} + \beta_{n}e^{t} \\
\end{gather}$$

$$\begin{gather}

P_{n}'(t) = -nA_{n}\sin(nt) + nB_{n}\cos(nt) + \beta_{n}e^{t} \\

$$\begin{gather}

\end{gather}$$

$$\begin{gather}
P(x,t) = \sum_{n=1}^{+\infty} \left[ (\alpha_{n}-\beta_{n})\cos(nt) -\frac{\beta_{n}}{n}\sin(nt) - \alpha_{n} + \beta_{n}e^{t} \right] \sin(nx)

Итого:

$$\begin{gather}
u = H + W + P \\
\end{gather}$$

## ТР 3

$$\begin{gather}
\begin{cases}
u_{tt} - \Delta u = 0 \\
u|_{t=0} = x^{2} - y^{2} \\
u_{t}|_{t=0} = xy^{3}
\end{cases}
\end{gather}$$

## Решение

$$\begin{gather}
\Delta \phi = \Delta(x^{2} - y^{2}) = 1 - 1 = 0 \implies \phi(x,y) - \text{гарм.}
\end{gather}$$

$$\begin{gather}
\Delta \psi = \Delta(xy^{3}) = 6xy \neq 0 \implies \psi(x,y) - \text{не гарм.}
\end{gather}$$

Замена:

$$\begin{gather}
u = v + \phi \\
\begin{cases}
v_{tt} - \Delta v = 0 \\
v|_{t=0} = 0 \\
v_{t}|_{t=0} = xy^{3}
\end{cases}
\end{gather}$$

$$\begin{gather}
v(x,y,t) = \frac{1}{2\pi}\iint_{r < t}  \, \frac{\psi(x,y)d\xi d\eta}{\sqrt{ t^{2}-r^{2}}}, \\
r^{2} = |\xi - x|^{2} + |\eta - y|^{2}
\end{gather}$$

$$\begin{gather}
\xi = r\cos \phi + x \\
\eta = r\sin \phi + y \\
\end{gather}$$

$$\begin{gather}
v = \frac{1}{2\pi}\int_{0}^{t} \frac{r dr}{\sqrt{ t^{2} - r^{2} }}\int_{0}^{2\pi} (r\cos \phi + x)(r\sin \phi + y)^{3} \, d\phi = \dots
\end{gather}$$

Рассмотрим подинтегральное выражение:

$$\begin{gather}
(r\cos \phi + x)(r^{3}\sin ^{3}\phi + 3yr^{2}\sin ^{2}\phi + 3y^{2}r\sin \phi + y^{3}) = \\
= r^{4}\underbrace{\cos \phi \sin ^{3}\phi}_{0} + 3yr^{3}\underbrace{\cos \phi \sin ^{2}\phi}_{0} + 3r^{2}y^{2}\underbrace{\sin \phi \cos \phi}_{0} + y^{3}r\underbrace{\cos \phi}_{0} + \\
+ xr^{3}\underbrace{\sin ^{3}\phi}_{0} + 3xyr^{2}\sin ^{2}\phi + 3xy^{2}r\underbrace{\sin \phi}_{0} + xy^{3} 
\end{gather}$$

Тогда внутренний интеграл принимает вид:

$$\begin{gather}
3xyr^{2}\int_{0}^{2\pi} \sin ^{2}\phi \, d\phi + xy^{3}\int_{0}^{2\pi} \, d\phi = \\
= \frac{3xyr^{2}}{2} \int_{0}^{2\pi} (1-\cos 2\phi) \, d\phi + 2\pi xy^{3} = \\
= 3\pi xyr^{2} + 2\pi xy^{3}
\end{gather}$$

$$\begin{gather}
\dots = \frac{3}{2}xy\int_{0}^{t} \underbrace{\frac{r^{3}}{\sqrt{ t^{2} - r^{2} }} \, dr}_{I_{1}} + xy^{3}\underbrace{\int_{0}^{t} \frac{r}{\sqrt{ t^{2} - r^{2} }} \, dt}_{I_{2}} = \dots
\end{gather}$$

$$\begin{gather}
I_{2} = \left.-\sqrt{ t^{2}-r^{2} } \right|_{0}^{t} = t
\end{gather}$$

$$\begin{gather}
I_{1} = \begin{bmatrix}
u = r^{2} \\
dv = \frac{r}{\sqrt{ t^{2} - r^{2} }}
\end{bmatrix} = \underbrace{\left.-r^{2}\sqrt{ t^{2} - r^{2} }\right|_{0}^{t}}_{0} + \int_{0}^{t} r\sqrt{ t^{2} - r^{2} } \, dr = \\
= \left.\frac{1}{2} \cdot \frac{2}{3} (t^{2} - r^{2})^{3/2}\right|_{0}^{t} = \frac{1}{3}t^{3}
\end{gather}$$

$$\begin{gather}
\dots = \frac{3}{2}xy \cdot \frac{1}{3}t^{3} + xy^{3} \cdot t = \frac{xyt^{3}}{2} + xy^{3}t
\end{gather}$$

Ответ:

$$\begin{gather}
u = \frac{xyt^{3}}{2} + xy^{3}t +  x^{2} - y^{2}
\end{gather}$$


## ТР 4

$$\begin{cases}
u_{tt} - u_{xx} - u_{yy} = t^{2}+y^{2}, \ t> 0, \ (x,y) \in (0,1) \times (0,2\pi) \\
u(x,y,0) = x(x-1)\sin \frac{y}{2}, \ u_{t}(x,y,0) = 0 \\
u(0,y,t) = u(1,y,t) = u(x,0,t) = u(x,2\pi,t) = 0
\end{cases}$$

### Решение

$$\begin{gather}
u = V + W
\end{gather}$$

$$\begin{cases}
V_{tt} - V_{xx} - V_{yy} = 0 \\
V(x,y,0) = x(x-1)\sin \frac{y}{2}, \ V_{t}(x,y,0) = 0 \\
V(0,y,t) = V(1,y,t) = V(x,0,t) = V(x,2\pi,t) = 0
\end{cases}$$

$$\begin{cases}
W_{tt} - W_{xx} - W_{yy} = t^{2}+y^{2} \\
W(x,y,0) = W_{t}(x,y,0) = 0 \\
W(0,y,t) = W(1,y,t) = W(x,0,t) = W(x,2\pi,t) = 0
\end{cases}$$

$$\begin{gather}
V = X(x)Y(y)T(t) \\
XYT'' - X''YT - XY''T = 0 \\
\frac{T''}{T} - \frac{X''}{X} - \frac{Y''}{Y} = 0 \\
\frac{T''}{T} = \frac{X''}{X} + \frac{Y''}{Y} = -\lambda^{2} = \mathrm{const}
\end{gather}$$

$$\begin{gather}
\frac{X''}{X} = -\frac{Y''}{Y} - \lambda^{2} = -\mu^{2}
\end{gather}$$

$$\begin{gather}
\begin{cases}
T'' + \lambda^{2}T = 0 \\
X'' + \mu^{2}X = 0 \\
Y'' + (\lambda^{2} - \mu^{2})Y = 0
\end{cases}
\end{gather}$$

$$\begin{gather}
X(x) = C_{1}\cos(\mu x) + C_{2}\sin(\mu x) \\
X(0) = C_{1} \cdot 1 = 0 \implies C_{1} = 0 \\
X(1) = C_{2}\sin(\mu) = 0 \implies \mu = \pi n \\
X_{n}(x) = \sin(\pi nx), \ n = 1,2,3\dots
\end{gather}$$

$$\begin{gather}
Y(y) = C_{1}\cos(\sqrt{ \lambda^{2}-\mu^{2} } \cdot y) + C_{2}\sin(\sqrt{ \lambda^{2}-\mu^{2} } \cdot y) \\
Y(0) = C_{1}\cdot 0 = 0 \implies C_{1} = 0 \\
Y(2\pi) = C_{1}\sin(\sqrt{ \lambda^{2}-\mu^{2} }\cdot 2\pi) = 0 \implies \sqrt{ \lambda^{2} - \mu^{2} } = 1 \implies \\
\implies \sqrt{ \lambda^{2} - \pi^{2}n^{2} } = \frac{m}{2} \\
\lambda^{2} - \pi^{2}n^{2} = \frac{m^{2}}{4} \\
\omega_{n,m} = \lambda_{n,m} = \sqrt{ \frac{m^{2}}{4} + \pi^{2}n^{2} }
\end{gather}$$

$$\begin{gather}
Y_{m}(y) = \sin\left( \frac{my}{2} \right), \ m=1,2,3\dots
\end{gather}$$

$$\begin{gather}
V = \sum_{n,m=1}^{+\infty} [A_{n,m}\sin(\omega_{n,m}t) + B_{n,m}\cos(\omega_{n,m}t)]\sin(\pi nx)\sin\left( \frac{my}{2} \right)
\end{gather}$$

$$\begin{gather}
V_{t} = \sum_{n,m=1}^{+\infty}\omega_{n,m}[A_{n,m}\cos(\omega_{n,m}t) - B_{n,m}\sin(\omega_{n,m}t)]\sin(\pi nx)\sin\left( \frac{my}{2} \right) \\
\end{gather}$$

$$\begin{gather}
V|_{t=0} = \sum_{n,m=1}^{+\infty}B_{n,m}\sin(\pi nx)\sin\left( \frac{my}{2} \right) = x(x-1)\sin \frac{y}{2} \\
V_{t}|_{t=0} = \sum_{n,m=1}^{+\infty}\omega_{n,m}A_{n,m}\sin(\pi nx)\sin\left( \frac{my}{2} \right) = 0 \implies A_{n,m} = 0
\end{gather}$$

$$\begin{gather}
B_{n,m} = \frac{\left( x(x-1)\sin \frac{y}{2}, \sin(\pi nx)\sin\left( \frac{my}{2} \right) \right)}{||\sin(\pi nx)\sin\left( \frac{my}{2} \right)||^{2}}
\end{gather}$$

$$\begin{gather}

$$\begin{gather}

= \pi \int_{0}^{1}x(x-1)\sin(\pi nx) \, dx = \\

$$\begin{gather}
I_{3} = \int_{0}^{1}x^{2}\sin(\pi nx) \, dx = \\

## ТР 5

$$\begin{gather}
\begin{cases}
\frac{1}{a^{2}}u_{tt} = u_{xx} + u_{yy} + x-t, \ x^{2} + y^{2} < r_{0}^{2}, \ t > 0 \\
u|_{x^{2}+y^{2} = r_{0}^{2}} = 0, \\
u|_{t=0} = J_{0}\left( \frac{\mu_{5}^{(0)}r}{r_{0}} \right), \ u_{t}|_{t=0} = 0
\end{cases}
\end{gather}$$

### 1. Переход к полярным координатам

$$\begin{gather}
\begin{cases}
x = r\cos \phi \\
y = r\sin \phi
\end{cases} \\
\Delta = \frac{1}{r} \frac{\partial }{\partial r}\left( r \frac{\partial}{\partial r} \right) + \frac{1}{r^{2}} \frac{\partial^{2}}{\partial \phi^{2}}
\end{gather}$$

$$\begin{cases}
\frac{1}{a^{2}}u_{tt} - \frac{1}{r}(ru_{r})_{r} - \frac{1}{r^{2}}u_{\phi \phi} = r\cos \phi - t, \ r < r_{0}, \ t > 0 \\
u|_{r=r_{0}} = 0 \\
u|_{t=0} = J_{0}\left( \frac{\mu_{5}^{(0)}r}{r_{0}} \right), \ u_{t}|_{t=0} = 0
\end{cases}$$

### 2. Разбиение на подзадачи

$$\begin{gather}
u = V + W
\end{gather}$$

$$\begin{cases}
\frac{1}{a^{2}}V_{tt} - \frac{1}{r}(rV_{r})_{r} \frac{1}{r^{2}}V_{\phi \phi} = 0 \\
V|_{r=r_{0}} = 0, \ \left( V_{\phi+2\pi} = V|_{\phi} \right), \\
V|_{t=0} = J_{0}\left( \frac{\mu_{5}^{(0)}r}{r_{0}} \right), \ V_{t}|_{t=0} = 0
\end{cases}$$

$$\begin{cases}
\frac{1}{a^{2}}W_{tt} - \frac{1}{r}(rW_{r})_{r} \frac{1}{r^{2}}W_{\phi \phi} = r\cos \phi - t, \\
W|_{r=r_{0}} = 0, \left( W_{\phi+2\pi} = W|_{\phi} \right), \\
W|_{t=0} = 0, \ W_{t}|_{t=0} = 0
\end{cases}$$

### 3. Решение для $V$

$$\begin{gather}
V = R(r)\Phi(\phi)T(t) \\
\frac{1}{a^{2}}R\Phi T'' - \frac{1}{r}(rR')'\Phi T - \frac{1}{r^{2}}R\Phi''T = 0 \\
\end{gather}$$

$$\begin{gather}
\frac{1}{a^{2}}\cdot \frac{T''}{T} - \frac{1}{rR}(rR')' - \frac{\Phi''}{r^{2}\Phi} = 0
\end{gather}$$

$$\begin{gather}
\frac{\Phi''}{\Phi} = r^{2} \cdot \frac{1}{a^{2}} \frac{T''}{T} - \frac{r}{R} (rR')' = -\xi^{2} = \mathrm{const} \\
\Phi'' + \xi^{2}\Phi = 0
\end{gather}$$

$$\begin{gather}
r^{2}\cdot \frac{1}{a^{2}} \frac{T''}{T} - \frac{r}{R}(rR')' = -\xi^{2} \\
\frac{1}{a^{2}}\cdot \frac{T''}{T} = \frac{1}{rR}(rR')' - \frac{\xi^{2}}{r^{2}} = -\eta^{2} = \mathrm{const} \\
\end{gather}$$

$$\begin{cases}
\Phi'' + \xi^{2}\Phi = 0 \\
r(rR')' + (\eta^{2}r^{2} - \xi^{2})R = 0 \\
T'' + \eta^{2}a^{2}T = 0
\end{cases}$$

$$\begin{gather}
\begin{cases}
\Phi'' + \xi^{2}\Phi = 0 \\
\Phi(\phi + 2\pi) = \Phi(\phi) \\
\end{cases} \\
\end{gather}$$

Используем известное решение:

$$\begin{gather}
\Phi_{n} = e^{in\phi}, \\
\xi_{n} = n, \ n \in \mathbb{Z}
\end{gather}$$

$$\begin{gather}
r^{2}R'' + rR' + (\eta^{2}r^{2} - n^{2})R = 0 \\
\end{gather}$$

Используем замену $x = \eta r$:

$$\begin{gather}
x^{2}R'' + xR' + (x^{2} - n^{2})R = 0 \\
\end{gather}$$

Тогда 

$$\begin{gather}
R = C_{1}\cdot J_{n}(x) + \cancel{ {C_{2}Y_{n}(x)} }
\end{gather}$$

В таком разбиении функция Бесселя в нуле ограничена, а фукнция Неймана - нет, потому мы eё выбрасываем.

$$\begin{gather}
R(r) = C_{1}J_{n}(\eta r) \\
R(r_{0}) = 0, \\
R(r_{0}) = C_{1}J_{n}(\eta r_{0}) = 0, \\
\eta r_{0} = \mu^{(n)}_{m} \\
\end{gather}$$

$$\begin{gather}
R_{n,m}(r) = J_{n}\left( \frac{\mu_{m}^{(n)}r}{r_{0}} \right), \ \eta_{m} = \frac{\mu_{m}^{(n)}}{r_{0}}, \ m = 1,2,3,\dots
\end{gather}$$

$$\begin{gather}

\end{gather}$$

$$\begin{gather}

\end{gather}$$

$$\begin{gather}

\end{gather}$$
