## Математические методы механики №1
### Ершов Алексей Геннадьевич КМБО-03-23

### Вариант 5

$$\begin{align}
& x = 2e^{t}, \ y = e^{2t} - 1 \\
& t_{1} = 1 \\
& \hline
\vec{v} - ?, \ \vec{a} - ?, \ \vec{a}_{\tau} - ? \\
& R(t) - ?, \ R(t)|_{t=t_{1}} - ?
\end{align}$$

$$\begin{gather}
x|_{t=t_{1}} = 2, \ y|_{t=t_{1}} = 0
\end{gather}$$

$$\begin{gather}
t = \ln \frac{x}{2} \\
y = e^{2 \ln x/2} - 1 \\
y = e^{\ln x^{2}/4} - 1 \\
y = \frac{x^{2}}{4} - 1 - \text{парабола} \\
\end{gather}$$

$$\begin{gather}
\dot{x} = 2e^{t} \\
\dot{y} = 2e^{2t} \\
\vec{v}(t) = (2e^{t}, 2e^{2t}) \\
v(t)|_{t=t_{1}} = (2e^{2}, 2e^{4}) \\
\end{gather}$$

$$\begin{gather}
v(t) = 2\sqrt{ e^{2t} + e^{4t} } \\
\vec{\tau} = \frac{\vec{v}}{|\vec{v}|} \frac{1}{\sqrt{ e^{2t} + e^{4t} }}(e^{t},e^{2t})
\end{gather}$$

$$\begin{gather}
\ddot{x} = 2e^{t}, \ \ddot{y} = 4e^{2t} \\
\vec{a} = (2e^{t}, 4e^{2t}) \\
\vec{a}|_{t=t_{1}} = (2e^{2}, 4e^{4}) \\
\end{gather}$$

$$\begin{gather}
\vec{a}_{\tau} = \frac{dv}{dt}\cdot \vec{\tau} = 2\frac{e^{2t} + 2e^{4t}}{\sqrt{ e^{2t} + e^{4t} }}\cdot \frac{1}{\sqrt{ e^{2t} + e^{4t} }} (e^{t}, e^{2t}) = \frac{2e^{2t}+4e^{4t}}{e^{2t}+e^{4t}}(e^{t}, e^{2t}) \\
\vec{a}_{\tau}|_{t=t_{1}} = \frac{2e^{4} + 4e^{8}}{e^{4} + e^{8}}(e^{2},e^{4})
\end{gather}$$

$$\begin{gather}
R(t) = \frac{|\vec{v}|^{3}}{|\vec{v} \times \vec{a}|} = \frac{8(e^{2t} + e^{4t})^{\frac{3}{2}}}{2e^{t}\cdot 4e^{2t} - 2e^{2t} \cdot 2e^{t}} = \frac{2(e^{2t} + e^{4t})^{\frac{3}{2}}}{e^{3t}} \\
R(t)|_{t=t_{1}} = \frac{2(e^{4} + e^{8})^{3/2}}{e^{6}}
\end{gather}$$
