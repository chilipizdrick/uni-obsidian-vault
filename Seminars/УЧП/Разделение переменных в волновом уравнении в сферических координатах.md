
$$\begin{gather}
\Delta u - \frac{1}{a^{2}} u_{tt} = 0 \\
\end{gather}$$

$$\begin{gather}
u = T(t)V(r, \theta, \phi)
\end{gather}$$

$$\begin{gather}
 T\Delta V - \frac{1}{a^{2}}V T'' = 0 \\
 \frac{\Delta V}{V} - \frac{1}{a^{2}}\frac{T''}{T} = 0 \\
 \frac{\Delta V}{V} = \frac{1}{a^{2}} \frac{T''}{T} = -k^{2} = \mathrm{const}
\end{gather}$$

$$\begin{gather}
\begin{cases}
T'' + (ka)^{2}T = 0 \\
\Delta V + k^{2}V = 0
\end{cases}
\end{gather}$$

$$\begin{gather}
\Delta V + k^{2}V = 0 \\
V = R(r)Y(\theta,\phi)
\end{gather}$$

где $Y(\theta, \phi) (Y_{l}^{m}(\theta, \phi))$ - сферические гармоники.

$$\begin{gather}
\Delta u = \frac{1}{r^{2}} \frac{ \partial  }{ \partial r } \left( r^{2} \frac{ \partial u }{ \partial r }  \right) + \frac{1}{r^{2}\sin \theta} \frac{ \partial  }{ \partial \theta } \left( \sin \theta \frac{ \partial u }{ \partial \theta }  \right) + \frac{1}{r^{2}\sin ^{2} \theta} \frac{ \partial^{2} u }{ \partial \phi^{2} } = \\
= \frac{\partial^{2} u}{\partial r^{2}} + \frac{2}{r} \frac{\partial u}{\partial r} + \frac{\cos \theta}{r^{2} \sin \theta} \frac{ \partial u }{ \partial \theta } + \frac{1}{r^{2}} \frac{ \partial^{2} u }{ \partial \theta^{2} } + \frac{1}{r^{2}\sin ^{2} \theta} \frac{ \partial^{2} u }{ \partial \phi^{2} }
\end{gather}$$

$$\begin{gather}
\Delta v = \frac{Y}{r^{2}} \left( r^{2} R_{r} \right)_{r} + \frac{R}{r^{2}\sin \theta} \left( \sin \theta Y_{\theta} \right)_{\theta} + \frac{R}{r^{2}\sin ^{2} \theta} Y_{\phi \phi} = \\
= \frac{1}{R}(r^{2}R_{r})_{r} + \frac{1}{Y\sin \theta}(\sin \theta Y_{\theta})_{\theta} + \frac{1}{Y\sin ^{2}\theta}Y_{\phi \phi}
\end{gather}$$

$$\begin{gather}
\frac{1}{R}(r^{2}R_{r})_{r} + k^{2}r^{2} + \frac{1}{Y\sin \theta}(\sin \theta Y_{\theta})_{\theta} + \frac{1}{Y\sin ^{2}\theta}Y_{\phi \phi} = 0 \\
\frac{1}{R}(r^{2}R_{r})_{r} + k^{2}r^{2} = -\frac{1}{Y\sin \theta}(\sin \theta Y_{\theta})_{\theta} - \frac{1}{Y\sin ^{2}\theta}Y_{\phi \phi} = l(l+1) = \mathrm{const} \\
l = 0,1,2\dots
\end{gather}$$

$$\begin{gather}
\begin{cases}
(r^{2}R_{r})_{r} + (k^{2}r^{2} - l(l+1))R = 0 \\
\frac{1}{\sin \theta}(\sin \theta Y_{\theta})_{\theta} + \frac{1}{\sin ^{2}\theta} Y_{\phi \phi} + l(l+1)Y = 0
\end{cases}
\end{gather}$$

Константа $l(l+1)$ здесь выбирается потому, что это спектр собственных значений оператора Лапласа на сфере, это необходимо для схождения решения к конечному на полюсах сферы.

$$\begin{gather}
Y(\theta, \phi) = \Theta(\theta)\Phi(\phi) \\
\end{gather}$$

$$\begin{gather}
\frac{\Phi}{\sin \theta}(\sin \theta \cdot \Theta' )' + \frac{\Theta}{\sin ^{2}\theta} \Phi'' + l(l+1)\Theta \Phi = 0 \\
\frac{\sin \theta}{\Theta}(\sin \theta \cdot \Theta')' + l(l+1)\sin ^{2}\theta = - \frac{\Phi''}{\Phi} = m^{2}, \ m \in \mathbb{Z}
\end{gather}$$

$$\begin{gather}
\begin{cases}
\Phi'' + m^{2}\Phi = 0 \\
\frac{1}{\sin \theta}(\sin \theta \cdot \Theta')' + \left(l(l+1) - \frac{m^{2}}{\sin ^{2} \theta} \right)\Theta = 0
\end{cases}
\end{gather}$$

$$\begin{gather}
\begin{cases}
T'' + (ka)^{2}T = 0 \\
(r^{2}R_{r})_{r} + (k^{2}r^{2} - l(l+1))R = 0 \\
\Phi'' + m^{2}\Phi = 0 \\
\frac{1}{\sin \theta}(\sin \theta \cdot \Theta')' + \left(l(l+1) - \frac{m^{2}}{\sin ^{2} \theta} \right)\Theta = 0
\end{cases}
\end{gather}$$

Заметив, что второе и четвертое уравнения - уравнение Бесселя и обобщенное уравнение Лежандра (где $x = \cos \theta$) соответственно, получаем решения.

Обобщенное уравнение Лежандра:
$$\begin{gather}
(1-x^{2}) y'' - 2xy' + \left[ l(l+1) - \frac{m^{2}}{1-x^{2}} \right]y = 0
\end{gather}$$

Приведение к виду происходит при использовании основного тригонометрического тождества:
$$\begin{gather}
\frac{1}{\sin \theta}(\sin \theta \cdot \Theta')' + \left(l(l+1) - \frac{m^{2}}{\sin ^{2} \theta} \right)\Theta = 0 \\
\frac{1}{\sin \theta}(\sin \theta \cdot \Theta')' + \left(l(l+1) - \frac{m^{2}}{1 - \cos^{2} \theta} \right)\Theta = 0
\end{gather}$$

Уравнение Бесселя:
$$\begin{gather}
x^{2}y'' + xy' + (x^{2} - k^{2})y = 0
\end{gather}$$

$$\begin{gather}
\begin{cases}
T(t) = A_{k} \cos(\omega t) + B_{k}\sin(\omega t), \ \omega = ka, \\
R(r) = C_{k,l} J_{l}(kr) + D_{k,l} Y_{l}(kr), \\
\Theta(\theta) = E_{m,l}P_{l}^{|m|}(\cos \theta), \\
\Phi(\phi) = F_{m}\cos(m\phi) + G_{m}\sin(m\phi),
\end{cases}
\end{gather}$$

гдe $J_{l}$ - фукнции Бесселя первого рода, $Y_{l}$ - функции Бесселя второго рода (функции Неймана), $P_{l}^{m}$ - присоединенные многочлены лежандра, $m$ - порядок, $l$ - степень.

$$\begin{gather}
P_{l}^{m}(x) = (-1)^{m}(1-x^{2})^{m/2} \frac{d^{m}}{d^{m} x} (P_{l}(x)) \\
P_{l}(x) = \frac{1}{2^{l}l!} \frac{d^{l}}{dx^{l}} \left[ (x^{2} - 1)^{l} \right] 
\end{gather}$$

$$u = R(r)\Theta(\theta)\Phi(\phi)T(t)$$
