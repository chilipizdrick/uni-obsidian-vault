### 1. Уравнения электродинамики для монохроматического электромагнитного поля

$$\begin{gather}
f(\vec{r}, t) = f_{m}(\vec{r})\cos(\omega t + \phi(\vec{r})); \ \omega = \frac{2\pi}{T} \\
f(\vec{r}, t) = \mathrm{Re}\dot{f}(\vec{r}, t) = \mathrm{Re} (\dot{\vec{f}_{m}}(\vec{r}) e^{i\omega t}) \\
f_{m}(\vec{r}) - \text{амплитуда} \\
\dot{\vec{f}}_{m} - \text{комплексная амплитуда} \\
\omega t + \phi(\vec{r}) - \text{фаза}
\end{gather}$$

Рассмотрим полную систему уравнений электродинамики для монохроматического электромагнитного поля в линейной изотропной среде.

$$\begin{align}
\begin{cases}
\mathrm{rot} \ \vec{H} = \epsilon_{a} \frac{\partial \vec{E}}{\partial t} + \sigma \vec{E} + \vec{j}^{ст} \\
\mathrm{rot} \ \vec{E} = - \mu_{a} \frac{\partial \vec{H}}{\partial t} \\
\mathrm{div} \ (\epsilon_{a}\vec{E}) = \rho \\
\mathrm{div}(\mu_{a}\vec{H}) = 0
\end{cases}
\end{align}$$


$$\begin{gather}
\vec{E}(\vec{r}, t) = \mathrm{Re}(\dot{\vec{E}}_{m}(\vec{r})e^{i\omega t}) \\
\vec{H}(\vec{r}, t) = \mathrm{Re}(\dot{\vec{H}}_{m}(\vec{r})e^{i\omega t}) \\
\vec{j}^{ст}(\vec{r}, t) = \mathrm{Re}(\dot{\vec{j}^{ст}}_{m}(\vec{r})e^{i\omega t}) \\
\end{gather}$$

$$\begin{gather}
\begin{cases}
\mathrm{rot} \ \dot{\vec{H}} = \epsilon_{a} \frac{\partial  \dot{\vec{E}}}{\partial t} + \sigma \dot{\vec{E}} + \dot{\vec{j}}^{ст}\\
\mathrm{rot} \ \dot{\vec{E}} = - \mu_{a} \frac{\partial \dot{\vec{H}}}{\partial t} \\
\mathrm{div} \ (\epsilon_{a}\dot{\vec{E}}) = \dot{\rho} \\
\mathrm{div}(\mu_{a}\dot{\vec{H}}) = 0
\end{cases}
\end{gather}$$

$$\begin{gather}
\begin{cases}
\mathrm{rot} \ \dot{\vec{H}}_{m} = i\omega\epsilon_{a}\dot{\vec{E}}_{m} + \sigma \dot{\vec{E}}_{m} + \dot{\vec{j}}^{ст}_{m}\\
\mathrm{rot} \ \dot{\vec{E}}_{m} = - i\omega\mu_{a}  \dot{\vec{H}}_{m} \\
\mathrm{div} \ (\epsilon_{a}\dot{\vec{E}}_{m}) = \dot{\rho}_{m} \\
\mathrm{div}(\mu_{a}\dot{\vec{H}}_{m}) = 0
\end{cases}
\end{gather}$$

### Уравнения электродинамики для однородной среды

$$\begin{gather}
\begin{cases}
\mathrm{rot} \ \dot{\vec{H}}_{m} = i\omega\epsilon_{a} \dot{\vec{E}}_{m} + \sigma \dot{\vec{E}}_{m} + \dot{\vec{j}}^{ст}_{m}\\
\mathrm{rot} \ \dot{\vec{E}}_{m} = - i\omega\mu_{a}  \dot{\vec{H}}_{m} \\
\mathrm{div} \ (\epsilon_{a}\dot{\vec{E}}_{m}) = \dot{\rho}_{m} \\
\mathrm{div}(\mu_{a}\dot{\vec{H}}_{m}) = 0
\end{cases}
\end{gather}$$

$$\begin{gather}
\dot{\epsilon}_{a} = \epsilon_{a}\left( 1-\frac{i\sigma}{\omega\epsilon_{a}} \right) \\
\mathrm{rot} \ \dot{\vec{H}}_{m} = i\omega \dot{\epsilon}_{a} \dot{\vec{E}}_{m} + \sigma \dot{\vec{E}}_{m} + \dot{\vec{j}}^{ст}_{m}\\
\end{gather}$$

$$\begin{gather}
\begin{cases}
0 = i\omega \dot{\epsilon}_{a} + \mathrm{div} \dot{\vec{E}} + \mathrm{div} \ \dot{\vec{j}}_{m}^{ст} \\
0 = -i\omega \mu_{a} \mathrm{div} \ \dot{\vec{H}}_{m} \\
\mathrm{div} \ (\dot{\vec{E}}_{m}) = -\frac{\mathrm{div} \dot{\vec{j}}_{m}}{i\omega\epsilon_{a}} \\
\mathrm{div}(\dot{\vec{H}}_{m}) = 0 
\end{cases}
\end{gather}$$

### 3. Волновые уравнения электродинамики

### Уравнение 1

$$\begin{gather}
\mathrm{rot} \ \mathrm{rot} \ \dot{\vec{H}}_{m} = i\omega\dot{\epsilon}_{a} \mathrm{rot} \ \dot{\vec{E}}_{m} + \mathrm{rot} \ \dot{\vec{j}}_{m}^{ст} \\
\mathrm{grad}\ \underbrace{\mathrm{div}\dot{\vec{H}}_{m}}_{0} - \Delta \dot{\vec{H}}_{m} = \underbrace{i\omega\dot{\epsilon}_{a}(-i\omega \mu_{a})}_{\omega^{2} \dot{\epsilon}_{a}\mu_{a} = \dot{k}^{2}} \dot{\vec{H}}_{m} + \mathrm{rot} \ \dot{\vec{j}}_{m}^{ст} \\
\dot{k}^{2} = \omega^{2}\dot{\epsilon}_{a}\mu_{a} - \text{волновое число (среды)} \\
\dot{k} = \omega \sqrt{ \dot{\epsilon}_{a}\mu_{a} } = \underbrace{k'}_{>0} - i\underbrace{k''}_{\geq 0}
\end{gather}$$

$$\begin{gather}
\Delta \dot{\vec{H}}_{m} + \dot{k}^{2}\dot{\vec{H}}_{m} = -\mathrm{rot} \ \dot{\vec{j}}_{m}^{ст} \ (5)
\end{gather}$$
