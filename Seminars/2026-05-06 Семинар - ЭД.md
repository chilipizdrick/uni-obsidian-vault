\dot{\vec{E}}_{m}(\vec{r}) = \begin{cases}
C_{l}^{(-)}\dot{\vec{E}}_{mo}^{(-)}(\vec{r}) = \frac{1}{N_{o}} \dot{I}_{m}^{ст} (\vec{l}, \dot{\vec{E}}_{mo}^{(+)}(\vec{r}_{0}))\dot{\vec{E}}_{mo}^{(-)}(\vec{r}), z < z_{0} \\
C_{l}^{(+)}\dot{\vec{E}}_{mo}^{(+)}(\vec{r}) = \frac{1}{N_{o}} \dot{I}_{m}^{ст} (\vec{l}, \dot{\vec{E}}_{mo}^{(-)}(\vec{r}_{0}))\dot{\vec{E}}_{mo}^{(+)}(\vec{r}), z > z_{0} \\

$$\begin{gather}

### 1. $N_{0}$

$$\begin{gather}
\dot{\vec{E}}_{mo}^{(\pm)}(\vec{r}) = \vec{y}^{0} \dot{E}_{0} \sin \left( \frac{\pi}{a} x \right) e^{\mp i \Gamma_{o} z} \\
\dot{\vec{H}}_{mo}^{(\pm)}(\vec{r}) = \frac{\dot{E}_{0}}{W_{0}} \left( \mp \vec{x}^{0} \sin \left( \frac{\pi}{a} x \right) + \vec{z}^{0} i \frac{\pi}{\Gamma_{o} a} \cos \left( \frac{\pi}{a} x \right) \right) e^{\mp i \Gamma_{o} z} \\
\end{gather}$$

$$\begin{gather}
N_{0} = \int_{S_{\perp}} \left( \left[ \dot{\vec{E}}_{m o}^{(+)}, \dot{\vec{H}}_{mo}^{(-) }\right] - \left[ \dot{\vec{E}}_{m o}^{(-)}, \dot{\vec{H}}_{mo}^{(+) }\right] \right) \vec{z}^{0} \, ds = \\
= \dot{E}_{0}\int_{S_{\perp}} \left( -2\left[ \vec{y}^{0} \sin\left( \frac{\pi}{a}x \right), \vec{x}^{0}\frac{1}{\dot{W}_{0}} \sin \left( \frac{\pi}{a} x \right)  \right]  \right) \vec{z}^{0} \, ds = \\
= -\frac{2 \dot{E}_{0}}{\dot{W}_{0}} \int_{S_{\perp}} \sin\left( \frac{\pi}{a} x \right) \sin \left( \frac{\pi}{a}x \right)  \, dx = \\
= -\frac{2\dot{E}_{0}}{\dot{W}_{0}} \int_{0}^{a} \int_{0}^{b} \sin ^{2} \left( \frac{\pi}{a} x \right) \, dx \, dy = -\frac{2\dot{E}_{0}}{\dot{W}_{0}} ab
\end{gather}$$
