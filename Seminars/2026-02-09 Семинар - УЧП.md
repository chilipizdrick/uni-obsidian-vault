# Метод Фурье для волнового уравнения - задача 2

Уравнение описывает движение струны.
Неоднородная часть уравнения описывает внешние воздействия.

$$\begin{gather}
u_{tt} - u_{xx} = x+3t, \ x \in (0,\pi), t > 0 \\
t(0,t) = 0, \ u(\pi,t) = 2t - t^{2} \\
u(x,0) = 0, u_{t}(x,0) = 1 + \cos(2x)
\end{gather}$$

$$\begin{gather}
u = H + V - \text{ЗАМЕНА} \\
H(x,t) = a(t) + x\cdot b(t) \\
H(0,t) = 0, H(\pi,t) = 2t - t^{2}
\end{gather}$$

$H(x,t) = a(t) + x\cdot b(t)$ - берет на себя граниченые условия

$$\begin{gather}
a = 0, \ \pi \cdot b(t) = 2t - t^{2} \\
b(t) = \frac{2t-t^{2}}{\pi}
\end{gather}$$

$$\begin{gather}
H = \frac{x}{\pi} \cdot (2t - t^{2}) \\
\begin{cases}
H_{tt} + V_{tt} - H_{xx} - V_{xx} = x + 3t \\
H(0,t) + V(0,t) = 0, \ H(\pi, t) + V(\pi ,t) = 2t - t^{2} \\
H(x,0) + V(x,0) = 0, \ H_{t}(x,0) + V_{t}(x,0) = 1 + \cos(2x)
\end{cases}
\end{gather}$$

$$\begin{gather}
\begin{cases}
V_{tt} - V_{xx} = x\left( 1+\frac{2}{\pi} \right) + 3t \\
V(0,t) = 0, \ V(\pi, t) = 0 \\
V(x,0) = 0, \ V_{t}(x,0) = 1+\cos(2x) - \frac{2x}{\pi}
\end{cases}
\end{gather}$$
