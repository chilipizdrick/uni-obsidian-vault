$$\begin{gather}
P_{A},P_{B}, \\
AB = l, \\
v_{0} = 0, \\
\omega = \mathrm{const} \\
\hline X_{\mathrm{max}}, \ Y_{\mathrm{max}} - ?
\end{gather}$$

Большие буквы $X,Y$ - реакции по оси $X$ и $Y$ соотв.

$$\begin{gather}
\phi = \omega t
\end{gather}$$

$$\begin{gather}
M \vec{a} = \sum \vec{F}_{k}^{e} \\
\end{gather}$$

1. $Ox$:

$$\begin{gather}
m_{A}x_{A} + m_{B}x_{B} = Mx_{c} \\
x_{A} = 0 \implies m_{B}x_{B} = Mx_{c} \\
x_{B} = l\sin \phi = l\sin\omega t
\end{gather}$$

$$\begin{gather}
m_{B}l\sin\omega t = Mx_{c} \\
M \ddot{x}_{c} = -m_{B}l\omega^{2}\sin \omega t
\end{gather}$$

$$\begin{gather}
\sum F_{kx}^{e} = X \\
X = -m_{B}l\omega^{2}\sin\omega t \\
X_{\mathrm{max}} = m_{B}l\omega^{2}
\end{gather}$$

2. $Oy$

$$\begin{gather}
m_{B}l\cos \omega t = My_{c} \\
M \ddot{y}_{c} = -m_{B}\omega^{2}\cos\omega t \\
\sum F_{ky}^{e} = Y - P_{A} - P_{B}
\end{gather}$$

$$\begin{gather}
Y - (P_{A} + P_{B}) = - m_{B}l\omega^{2}\cos\omega t \\
Y = (P_{A} + P_{B}) - m_{B}l\omega^{2}\cos\omega t \\
Y_{\mathrm{max}} = (P_{A} + P_{B}) + m_{B}l\omega^{2}
\end{gather}$$

### Задача 2

![[2026-03-25 Семинар - Механика 2026-03-25 14.58.41.excalidraw]]

$$\begin{gather}
P_{1}, P_{2}, P_{3} \\
OA = AB = r \\
\omega = \mathrm{const} \\
\hline X(\phi), Y(\phi) - ?
\end{gather}$$

Решение:

$$\begin{gather}
M \vec{a}_{c} = \sum \vec{F}_{k}^{e} \\
\end{gather}$$

1. $Ox$

$$\begin{gather}
m_{1}x_{1} + m_{2}x_{2} + m_{3}x_{3} = Mx_{c} \\
x_{1} = \frac{r}{2}\cos \phi \\
x_{2} = \frac{3r}{2}\cos \phi \\
x_{3} = 2r\cos \phi = \frac{4r}{2}\cos \phi
\end{gather}$$

$$\begin{gather}
\left(\frac{m_{1}r}{2} + \frac{3m_{2}r}{2} + \frac{4m_{3}r}{2}\right)\cos \phi = M x_{c}
\end{gather}$$

$$\begin{gather}
\frac{r}{2}\left( m_{1} + 3m_{2} + 4m_{3} \right) \cos \phi = Mx_{c}
\end{gather}$$

$$\begin{gather}
M \ddot{x}_{c} = -\frac{r\omega^{2}}{2}\left( m_{1} + 3m_{2} + 4m_{3} \right) \cos \phi \\
\sum F_{kx}^{e} = X \\
X = - \frac{r\omega^{2}}{2}(m_{1} + 3m_{2} + 4m_{3})\cos \phi
\end{gather}$$

2. $Oy$

$$\begin{gather}
y_{1} = \frac{r}{2}\sin \phi \\
y_{2} = \frac{r}{2}\sin \phi \\
y_{3} = 0
\end{gather}$$

$$\begin{gather}
\left(\frac{m_{1}r}{2} + \frac{m_{2}r}{2}\right)\sin \phi = My_{c}
\end{gather}$$

$$\begin{gather}
M \ddot{y}_{c} = -\frac{r\omega^{2}}{2}(m_{1} + m_{2}) \sin \phi \\
\sum F^{2}_{ky} = Y - (P_{1} + P_{2} + P_{3})
\end{gather}$$

$$\begin{gather}
Y - \left( P_{1} + P_{2} + P_{3}  \right) = -\frac{r\omega^{2}}{2}(m_{1} + m_{2})\sin \phi \\
Y = P_{1} + P_{2} + P_{3} -\frac{r\omega^{2}}{2}(m_{1} + m_{2})\sin \phi \\
\end{gather}$$

$$\begin{gather}
Y_{\mathrm{max}} = P_{1} + P_{2} + P_{3} + \frac{r\omega^{2}}{2}(m_{1} + m_{2})
\end{gather}$$
