### Пр. 6. 4.4 Полное прохождение ПОВ через границу раздела двух электрическиих сред (угол Брюстера).

![[2026-03-18 Семинар - ЭД 2026-03-18 12.46.16.excalidraw]]

$$\begin{gather}
k_{1}\sin \phi = k_{2}\sin \theta \to \sin \theta = \frac{k_{1}}{k_{2}}\sin \phi \ (1)
\end{gather}$$

$$\begin{gather}
\rho_{\parallel} = \frac{W_{1}\cos \phi - W_{2}\cos \theta}{W_{2}\cos \phi + W_{2}\cos \theta} = 0 \to (2)
\end{gather}$$

$$\begin{gather}
W_{1}\cos \phi = W_{2}\cos \theta \to \cos \phi = \frac{W_{2}}{W_{1}}\sqrt{ 1-\sin ^{2}\theta } \to \\
\to\cos \phi = \frac{W_{2}}{W_{1}}\sqrt{ 1 - \left( \frac{k_{1}}{k_{2}}\sin \theta \right)^{2} } (3)
\end{gather}$$

$$\begin{gather}
\frac{W_{2}}{W_{1}} = \left.\frac{\sqrt{ \mu_{a2} / \epsilon_{a 2} }}{\mu_{a 1} / \epsilon_{a 1}}\right|_{\mu_{a 2} = \mu_{a 1}} = \sqrt{ \frac{\epsilon_{1}}{\epsilon_{2}} }, \ \frac{k_{1}}{k_{2}} = \frac{\omega \sqrt{ \epsilon_{a 1}\mu_{a 1} }}{\omega \sqrt{ \epsilon_{a 2}\mu_{a 1} }} = \sqrt{ \frac{\epsilon_{1}}{\epsilon_{2}} } \ (4)\ \text{в} \ (3)
\end{gather}$$

$$\begin{gather}
\cos ^{2}\phi = \frac{\epsilon_{1}}{\epsilon_{2}}\left( 1 - \frac{\epsilon_{1}}{\epsilon_{2}}\sin ^{2}\phi \right) \to \\
\to 1 - \sin ^{2}\phi = \frac{\epsilon_{1}}{\epsilon_{2}} - \left( \frac{\epsilon_{1}}{\epsilon_{2}} \right)^{2}\sin ^{2}\phi \to \\
\to \sin ^{2}\phi = \frac{\epsilon_{2}}{\epsilon_{2} + \epsilon_{1}} \ (5)
\end{gather}$$

$$\begin{gather}
\cos ^{2}\phi = 1 - \sin ^{2}\phi = 1-\frac{\epsilon_{2}}{\epsilon_{2} + \epsilon_{1}} = \frac{\epsilon_{1}}{\epsilon_{2} - \epsilon_{1}} \ (6)
\end{gather}$$

Из $(5), (6) \to$

$$\begin{gather}
\frac{\sin^{2}\phi}{\cos ^{2}\phi} = \frac{\epsilon_{2}}{\epsilon_{1}} \to \\
\to \mathrm{tg} \ \phi = \sqrt{ \frac{\epsilon_{2}}{\epsilon_{1}} } \to \ \phi = \phi_{Б} = \mathrm{arctg} \ \sqrt{ \frac{\epsilon_{2}}{\epsilon_{1}} } \ (7)
\end{gather}$$
