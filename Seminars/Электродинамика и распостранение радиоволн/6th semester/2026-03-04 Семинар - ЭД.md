### Пр. №4. Нормальное падение ПОВ на бесконечную идеальную проводящую плоскость.

### 1. Постановка задачи

![[2026-03-04 Семинар - ЭД 2026-03-04 12.43.54.excalidraw]]

Дано: $\dot{\vec{E}}_{m}^{(0)} = \vec{x}^{0} \dot{E}_{0} e^{i\dot{k}z}$

Определить: $\dot{\vec{E}}_{m}(\vec{r}), \ \dot{\vec{H}}_{m}(\vec{r})$


$$\begin{gather}
\dot{\vec{E}}_{m\tau}|_{z=0} = 0 \ (*)
\end{gather}$$

### 2. Решение задачи

$$\begin{gather}
\dot{\vec{E}}_{m}(\vec{r}) = \dot{\vec{E}}_{m}^{(0)} + \dot{\vec{E}}_{m}^{вт}
\end{gather}$$

$$\begin{cases}
\dot{\vec{E}}_{m}(\vec{r}) = \frac{1}{i\omega\dot{\epsilon}_{a}} (\mathrm{grad}\, \mathrm{div}\, \dot{\vec{A}}_{m} + \dot{k}^{2} \dot{\vec{A}}_{m}) \\
\dot{\vec{H}}_{m}(\vec{r}) = \mathrm{rot}\, \dot{\vec{A}}_{m}
\end{cases}$$

$$\begin{gather}
\dot{\vec{A}}_{m}(\vec{r}) = \int_{S(z=0)} \dot{\vec{\eta}}_{m}(\vec{r})G(\vec{r}, \vec{r}') \, dv' \ (4)
\end{gather}$$

$$\begin{gather}
G(\vec{r}, \vec{r}') = \frac{e^{-ik|\vec{r}-\vec{r}'|}}{4\pi|\vec{r}-\vec{r}'|} \\
|\vec{r}-\vec{r}'| = \left.\sqrt{ (x-x')^{2} + (y-y')^{2} + (z-z')^{2} }\right|_{z'=0} = \sqrt{ (x-x')^{2} + (y-y')^{2} + z^{2} }
\end{gather}$$

Из-за плоскостной симметрии $\dot{\vec{\eta}}_{m}^{ст} = \mathrm{const}$

$$\begin{gather}
\dot{\vec{A}}_{m}(\vec{r}) = \dot{\vec{\eta}}_{m}^{ст}\int_{S(z=0)} \frac{e^{-ik}\sqrt{ (x')^{2} + (y')^{2} + z^{2} }}{4\pi\sqrt{ (x')^{2} + (y')^{2} + z^{2} }} \, ds'
\end{gather}$$

Перейдем к полярным координатам

$$\begin{gather}
\dot{\vec{A}}_{m}(\vec{r}) = \dot{\vec{\eta}}_{m}^{ст}\cdot \frac{1}{4\pi} \int_{\alpha'=0}^{2\pi} \int_{r=0}^{\infty} \frac{e^{-i\dot{k}\sqrt{ (r')^{2} + z^{2} }}}{\sqrt{ (r')^{2} + z^{2} }} r' \, dr'd\alpha' = 
\end{gather}$$

$$\begin{gather}
= \dot{\vec{\eta}}_{m}^{ст}\frac{1}{4\pi}\cdot 2\pi \int_{r'=0}^{\infty} \frac{e^{-i\dot{k}\sqrt{ (r')^{2} + z^{2} }}}{\sqrt{ (r')^{2} + z^{2} }} r' \, dr' = \\
= \dot{\vec{\eta}}_{m}^{ст} \frac{1}{2}\int_{r'=0}^{\infty} e^{-i\dot{k} \sqrt{ (r')^{2} + z^{2} }} \, d\sqrt{ (r')^{2} + z^{2} } =
\end{gather}$$

$$\begin{gather}
= \dot{\vec{\eta}}_{m}^{ст} \cdot \frac{1}{2} \int_{\xi=|z|}^{\infty} e^{-i\dot{k}\xi} \, d\xi = \\
= \left.\dot{\vec{\eta}}_{m}^{ст} \cdot \frac{1}{2} \frac{1}{-i\dot{k}}e^{-ik\xi} \right|_{|z|}^{\infty} = \frac{\dot{\vec{\eta}}_{m}^{ст}}{2i\dot{k}}e^{-i\dot{k}|z|}
\end{gather}$$

$$\begin{gather}
\dot{\vec{A}}_{m}(z) = \frac{\dot{\vec{\eta}}_{m}^{ст}}{2i\dot{k}}e^{-i\dot{k}|z|}
\end{gather}$$


$$\begin{gather}
\dot{\vec{\eta}}_{m}^{ст} = \vec{x}^{0} \dot{\eta}_{mx} + \vec{y}^{0}\dot{\eta}_{my}
\end{gather}$$

$$\begin{gather}
\mathrm{div}\, \dot{\vec{A}}_{m} = 0 \implies \mathrm{grad} \, \mathrm{div} \dot{\vec{A}}_{m} = 0
\end{gather}$$

$$\begin{gather}
\dot{\vec{E}}_{m} = \frac{1}{i\omega\dot{\epsilon}_{a}}(\mathrm{grad} \, \mathrm{div} \, \dot{\vec{A}}_{m} + \dot{k}^{2} \dot{\vec{A}}_{m}) = \frac{\dot{k}^{2}}{i\omega\dot{\epsilon}_{a}} \dot{\vec{A}}_{m} = \\
= -\dot{\vec{\eta}}_{m}^{ст}\frac{\dot{W}}{2}e^{-i\dot{k}|z|}
\end{gather}$$

$$\begin{gather}
\dot{\vec{E}}_{m}(z) = -(\vec{x}^{0}\dot{\eta}_{mx} + \vec{y}^{0}\dot{\eta}_{my}) \frac{\dot{W}}{2} e^{-ik|z|} \\
\dot{\vec{H}}_{m} = \frac{\mathrm{rot} \, \dot{\vec{E}}_{m}}{-i\omega\mu_{a}}
\end{gather}$$

$$\begin{gather}
\dot{\vec{E}}_{m} = \dot{\vec{E}}_{m}^{(0)} + \dot{\vec{E}}_{m}^{вт} = \vec{x}^{0}\dot{E}_{0}e^{ikz} -(\vec{x}^{0}\dot{\eta}_{mx} + \vec{y}^{0}\dot{\eta}_{my}) \frac{\dot{W}}{2} e^{-ik|z|}
\end{gather}$$

Из $(*) \implies \dot{\eta}_{my} = 0$.

$$\begin{gather}
-\vec{x}^{0} \dot{E}_{0} - \vec{x}^{0} \frac{\dot{\eta}_{mx}}{2} = 0 \implies \dot{\eta}_{mx} = \dot{\eta}_{m} = \frac{2\dot{E}_{0}}{\dot{W}}
\end{gather}$$

Тогда

$$\begin{gather}
\dot{\vec{E}}_{m} = \vec{x}^{0} \dot{E}_{0}e^{i\dot{k}z} + \begin{cases}
-\vec{x}^{0}\dot{E}_{0}e^{-i\dot{k}z}, \ z > 0 \\
-\vec{x}^{0}\dot{E}_{0}e^{i\dot{k}z}, \ z < 0
\end{cases} \\
\dot{\vec{E}}_{m}= \begin{cases}
\vec{x}^{0}\dot{E}_{0}i2\sin(\dot{k}z), \ z > 0 \\
0, \ z < 0
\end{cases}
\end{gather}$$
