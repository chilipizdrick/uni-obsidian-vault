## Пр. №14. Возбуждение прямоугольного волновода ЭЭИ.

![[2026-05-13 Семинар - ЭД 2026-05-13 13.30.13.excalidraw]]

$$\begin{gather}
\left.\dot{\vec{E}}_{m\tau}\right|_{z=0} = 0 \ (*)
\end{gather}$$

1. Бесконечный волновод

$$\begin{gather}
\dot{\vec{E}}_{m} = \vec{y}^{0} \dot{E}_{o}^{ст} \sin\left( \frac{\pi}{a} x_{0} \right) \sin\left( \frac{\pi}{a} x \right) e^{-i \Gamma_{o} |z-z_{0}|} \\
\dot{\vec{H}}_{m} = \frac{\dot{E}_{o}^{ст}}{\dot{W}_{o}} \sin\left( \frac{\pi}{a} x_{0} \right)\left( \pm \vec{x}^{0} \sin \left( \frac{\pi}{a}x \right) + \vec{z}^{0}i \frac{\pi}{\Gamma_{o}a} \cos \left( \frac{\pi}{a} x \right) \right) e^{-i \Gamma_{o} |z-z_{o}|}
\end{gather}$$

$$\begin{gather}
\Gamma_{o} = k \sqrt{ 1- \left( \frac{\lambda}{2a} \right)^{2} } \\
W_{o} = \frac{\omega \mu_{a}}{\Gamma_{o}} = \frac{W}{\sqrt{ 1- \left( \frac{\lambda}{2a} \right)^{2} }}, \ W = \sqrt{ \frac{\mu_{a}}{\epsilon_{a}} }
\end{gather}$$

2. Полубесконечный волновод

$$\begin{gather}
\dot{\vec{E}}_{m} = \dot{\vec{E}}_{m} ^{(0)} + \dot{\vec{E}}_{m}^{(-)} = \\
= \vec{y}^{0} \dot{E}_{o}\sin \left( \frac{\pi}{a} x_{0} \right) \sin \left( \frac{\pi}{a} x \right) \left( e^{-i\Gamma_{o}|z-z_{0}|} + \rho e^{-i\Gamma_{o}z} \right) \to \text{в} (*)
\end{gather}$$

$$\begin{gather}
e^{-i\Gamma_{o}z_{0}} + \rho = 0 \implies \rho = -e^{-i\Gamma_{o}z_{0}}
\end{gather}$$

$$\begin{gather}
\dot{\vec{E}}_{m} = \vec{y}^{0} \dot{E}_{0}^{ст} \sin\left( \frac{\pi}{a} x \right) \sin \left( \frac{\pi}{a} x_{0} \right) \begin{cases}
e^{-i\Gamma_{o} z_{0}}e^{i\Gamma_{o}z}  -e^{-i\Gamma_{o}z_{0}}e^{-i\Gamma_{o}z}, \ z < z_{0}, \\
e^{i\Gamma_{o} z_{0}}e^{-i\Gamma_{o}z} -e^{-i\Gamma_{o}z_{0}}e^{-i\Gamma_{o}z}, \ z \geq z_{0}.
\end{cases} \\
\end{gather}$$

$$\begin{gather}
\dot{\vec{E}}_{m} = \vec{y}^{0} 2i \dot{E}_{0}^{ст} \sin \left( \frac{\pi}{a}x \right)\sin\left( \frac{\pi}{a}x_{0} \right) \begin{cases}
e^{-i\Gamma_{o}z_{0}} \sin (\Gamma_{o}z), \ z < z_{0}, \\
e^{-i\Gamma_{o}z} \sin (\Gamma_{o}z_{0}), \ z \geq z_{0}.
\end{cases}
\end{gather}$$

Также решение методом зеркальных отображений:

$$\begin{gather}
\dot{\vec{E}}_{m} = \dot{\vec{E}}_{m}^{П} + \dot{\vec{E}}_{m}^{Вт} = \\
= \vec{y}^{0} \sin \left( \frac{\pi}{a} \right)\sin\left( \frac{\pi}{a} x_{0} \right) \begin{cases}
e^{-i\Gamma_{o} (z_{0}-z)} - e^{-i\Gamma_{o} (z+z_{0})}, z < z_{0} \\
e^{-i\Gamma_{o}(z-z_{0})} - e^{-i\Gamma_{o} (z+z_{0})}, z > z_{0}
\end{cases}
\end{gather}$$

$$\begin{gather}
\dot{\vec{E}}_{m} = \vec{y}^{0} i 2\sin \left( \frac{\pi}{a} \right)\sin\left( \frac{\pi}{a} x_{0} \right) \begin{cases}
e^{-i\Gamma_{o}z_{0}}\sin(\Gamma_{o}z), \ z < z_{0} \\
e^{-i\Gamma_{o}z}\sin (\Gamma_{o}z_{0}) , \ z > z_{0}
\end{cases}
\end{gather}$$

3. Возбуждение основного поля прямоугольного волновода, ограниченного с обоих концов идеально проводящими плоскостями.

Решается все аналогично первому методу, но коэффициентов отражения теперь два.

$$\begin{gather}
\dot{\vec{E}}_{m} = \vec{y}^{0} \dot{E}_{o}^{ст} \sin \left( \frac{\pi}{a} x_{0} \right)\sin \left( \frac{\pi}{a} x \right) (e^{-i\Gamma_{o}|z-z_{0}|} + \rho_{1}e^{-i\Gamma_{o}z} + \rho_{2} e^{i\Gamma_{o} z})
\end{gather}$$

Решение:

$$\begin{gather}
\dot{\vec{E}}_{m} =  \vec{y}^{0} \frac{i2}{\sin(\Gamma_{o} L)} \dot{E}_{o}^{ст} \sin\left( \frac{\pi}{a} x \right)\sin \left( \frac{\pi}{a} x_{0} \right) \begin{cases}
\sin[\Gamma_{o}(L-z_{0})] \sin (\Gamma_{o}z), \ z < z_{0}, \\
\sin(\Gamma_{o}z_{0})\sin[\Gamma_{o}(L-z)], \ z > z_{0}.
\end{cases}
\end{gather}$$

Запишем выражение для поля, когда $\Gamma_{o} L = \pi, \Gamma_{o} = \frac{\pi}{L}$:

$$\begin{gather}
\dot{\vec{E}}_{m} = \vec{y}^{0} \frac{i2}{\sin(\Gamma_{o}L)} \dot{E}_{o}^{ст} \sin \left( \frac{\pi}{a}x_{0} \right) \sin\left( \frac{\pi}{a} x \right) \begin{cases}
\sin(\Gamma_{o}z_{0}) \sin(\Gamma_{o}z), \ z < z_{0} \\
\sin(\Gamma_{o}z_{0}) \sin(\Gamma_{o}z), \ z > z_{0}
\end{cases}
\end{gather}$$

$$\begin{gather}
\dot{\vec{E}}_{m} \vec{y}^{0} i \frac{2\dot{E}_{o}^{ст}}{\sin\left( \frac{\pi}{L} L \right)} \sin\left( \frac{\pi}{a} x_{0} \right) \sin\left( \frac{\pi}{L} z_{0} \right) \sin\left( \frac{\pi}{a} x \right) \sin \left( \frac{\pi}{L} z \right)
\end{gather}$$

### 2

### Нахождение $\dot{\vec{E}}_{m}$ для прямоугольного резонатора:

Основное уравнение поля с учетом падающих и отраженных волн:
$$\dot{\vec{E}}_{m} = \vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \left(e^{-i\Gamma_{o}|z-z_{0}|} + \rho_{1}e^{-i\Gamma_{o}z} + \rho_{2}e^{i\Gamma_{o}z}\right)$$

* $\dot{\vec{E}}_{m\tau}|_{z=0} = 0$
* $\dot{\vec{E}}_{m\tau}|_{z=L} = 0$

---


$$\dot{\vec{E}}_{m\tau}|_{z=0} = 0 \Rightarrow \left(e^{-i\Gamma_{o}|z-z_{0}|} + \rho_{1}e^{-i\Gamma_{o}z} + \rho_{2}e^{i\Gamma_{o}z}\right)|_{z=0} = 0$$
$$e^{-i\Gamma_{o}z_{0}} + \rho_{1}e^{0} + \rho_{2}e^{0} = 0$$

$$\dot{\vec{E}}_{m\tau}|_{z=L} = 0 \Rightarrow \left(e^{-i\Gamma_{o}|z-z_{0}|} + \rho_{1}e^{-i\Gamma_{o}z} + \rho_{2}e^{i\Gamma_{o}z}\right)|_{z=L} = 0$$
$$e^{-i\Gamma_{o}(L-z_{0})} + \rho_{1}e^{-i\Gamma_{o}L} + \rho_{2}e^{i\Gamma_{o}L} = 0$$

$$\begin{cases} \rho_{1} + \rho_{2} = -e^{-i\Gamma_{o}z_{0}} \\ \rho_{1}e^{-i\Gamma_{o}L} + \rho_{2}e^{i\Gamma_{o}L} = -e^{-i\Gamma_{o}(L-z_{0})} \end{cases}$$

$$\rho_{1} = -e^{-i\Gamma_{o}z_{0}} - \rho_{2}$$

$$\left(-e^{-i\Gamma_{o}z_{0}} - \rho_{2}\right)e^{-i\Gamma_{o}L} + \rho_{2}e^{i\Gamma_{o}L} = -e^{-i\Gamma_{o}(L-z_{0})}$$
$$\rho_{2}\left(-e^{-i\Gamma_{o}L} + e^{i\Gamma_{o}L}\right) = e^{-i\Gamma_{o}z_{0}-i\Gamma_{o}L} - e^{-i\Gamma_{o}(L-z_{0})}$$

$$2i\rho_{2}\sin(\Gamma_{o}L) = -e^{-i\Gamma_{o}L} 2i \sin(\Gamma_{o}z_{0})$$
$$\rho_{2}\sin(\Gamma_{o}L) = -e^{-i\Gamma_{o}L}\sin(\Gamma_{o}z_{0})$$

$$\rho_{2} = -e^{-i\Gamma_{o}L}\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)}$$
$$\rho_{1} = -e^{-i\Gamma_{o}z_{0}} + e^{-i\Gamma_{o}L}\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)}$$

$$\dot{\vec{E}}_{m} = \vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \left[ e^{-i\Gamma_{o}|z-z_{0}|} + \left(-e^{-i\Gamma_{o}z_{0}} + e^{-i\Gamma_{o}L}\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)}\right)e^{-i\Gamma_{o}z} - \left(e^{-i\Gamma_{o}L}\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)}\right)e^{i\Gamma_{o}z} \right]$$

$$\dots = e^{-i\Gamma_{o}|z-z_{0}|} - e^{-i\Gamma_{o}(z+z_{0})} + e^{-i\Gamma_{o}(L+z)}\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)} - e^{-i\Gamma_{o}(L-z)}\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)}$$
$$\dots = e^{-i\Gamma_{o}|z-z_{0}|} - e^{-i\Gamma_{o}(z+z_{0})} + e^{-i\Gamma_{o}L}\left(e^{-i\Gamma_{o}z} - e^{i\Gamma_{o}z}\right)\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)}$$
$$\dots = e^{-i\Gamma_{o}|z-z_{0}|} - e^{-i\Gamma_{o}(z+z_{0})} - 2ie^{-i\Gamma_{o}L}\sin(\Gamma_{o}z)\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)}$$

$$\dot{\vec{E}}_{m} = \vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \left[ e^{-i\Gamma_{o}|z-z_{0}|} - e^{-i\Gamma_{o}(z+z_{0})} - 2ie^{-i\Gamma_{o}L}\sin(\Gamma_{o}z)\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)} \right]$$

---

### $z > z_{0}$:
$$\dot{\vec{E}}_{m} = \vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \left[ e^{-i\Gamma_{o}(z-z_{0})} - e^{-i\Gamma_{o}(z+z_{0})} - 2ie^{-i\Gamma_{o}L}\sin(\Gamma_{o}z)\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)} \right]$$

$$\dots = 2ie^{-i\Gamma_{o}z}\sin(\Gamma_{o}z_{0}) - 2ie^{-i\Gamma_{o}L}\sin(\Gamma_{o}z)\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)}$$

$$\dot{\vec{E}}_{m} = 2i\vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \sin(\Gamma_{o}z_{0}) \left[ e^{-i\Gamma_{o}z} - e^{-i\Gamma_{o}L}\frac{\sin(\Gamma_{o}z)}{\sin(\Gamma_{o}L)} \right]$$
$$\dot{\vec{E}}_{m} = 2i\vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)} \left[ e^{-i\Gamma_{o}z}\sin(\Gamma_{o}L) - e^{-i\Gamma_{o}L}\sin(\Gamma_{o}z) \right]$$

$$\dot{\vec{E}}_{m} = 2i\vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)} \sin(\Gamma_{o}(L-z))$$

---

### $z < z_{0}$:
$$\dot{\vec{E}}_{m} = \vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \left[ e^{i\Gamma_{o}(z-z_{0})} - e^{-i\Gamma_{o}(z+z_{0})} - 2ie^{-i\Gamma_{o}L}\sin(\Gamma_{o}z)\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)} \right]$$

$$\dots = e^{-i\Gamma_{o}z_{0}}\left(e^{i\Gamma_{o}z} - e^{-i\Gamma_{o}z}\right) - 2ie^{-i\Gamma_{o}L}\sin(\Gamma_{o}z)\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)}$$
$$\dot{\vec{E}}_{m} = 2i\vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \left[ e^{-i\Gamma_{o}z_{0}}\sin(\Gamma_{o}z) - e^{-i\Gamma_{o}L}\sin(\Gamma_{o}z)\frac{\sin(\Gamma_{o}z_{0})}{\sin(\Gamma_{o}L)} \right]$$

$$\dot{\vec{E}}_{m} = 2i\vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \frac{\sin(\Gamma_{o}z)}{\sin(\Gamma_{o}L)} \left[ e^{-i\Gamma_{o}z_{0}}\sin(\Gamma_{o}L) - e^{-i\Gamma_{o}L}\sin(\Gamma_{o}z_{0}) \right]$$

$$\dot{\vec{E}}_{m} = 2i\vec{y}^{0} E_{0}^{cr} \sin\left(\frac{\pi}{a}x_{0}\right) \sin\left(\frac{\pi}{a}x\right) \frac{\sin(\Gamma_{o}z)}{\sin(\Gamma_{o}L)} \sin(\Gamma_{o}(L-z_{0}))$$

---

### Итоговый ответ:
$$\dot{\vec{E}}_{m} = \vec{y}^{0} E_{o}^{ст} \sin\left(\frac{\pi}{a}x_{0}\right) \frac{2i\sin\left(\frac{\pi}{a}x\right)}{\sin(\Gamma_{o}L)} \begin{cases} \sin(\Gamma_{o}z)\sin(\Gamma_{o}(L-z_{0})), & z < z_{0} \\ \sin(\Gamma_{o}z_{0})\sin(\Gamma_{o}(L-z)), & z > z_{0} \end{cases}$$
