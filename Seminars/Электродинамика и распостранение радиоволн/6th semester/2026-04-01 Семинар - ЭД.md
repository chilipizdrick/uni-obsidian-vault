### Пр. №8. Диффракция ПОВ на отверстии в бесконечном идеально проводящем экране

$$\begin{gather}
\begin{cases}
\dot{\vec{E}}_{m} = \vec{x}^{0} \dot{E}_{0}e^{-ikz}, \\
\dot{\vec{H}}_{m} = \vec{y}^{0} \frac{\dot{E}_{0}}{\dot{W}}e^{-ikz}.
\end{cases}
\end{gather}$$

$$\begin{gather}
\left.\dot{\vec{E}}_{m}\right|_{z=0} \approx \vec{x}^{0} \dot{E}_{0}, \ \left.\dot{\vec{E}}_{m\tau} \right|_{S(z=+0)} \approx 0 \\
\left.\dot{\vec{H}}_{m}\right|_{z=0} \approx \vec{y}^{0} \dot{E}_{0}, \ \left.\dot{\vec{H}}_{m\tau} \right|_{S(z=+0)} \approx 0 
\end{gather}$$

$$\begin{gather}
\dot{\vec{E}}_{M}(\vec{r}) = \int_{S_{0}}  \dot{\vec{dE}}_{m}
\end{gather}$$

$$\begin{gather}

\dot{\vec{dE}}_{m} = ik(i+\cos \theta)(\vec{\theta}^{0}\cos\alpha - \vec{\alpha}^{0}\sin\alpha)G(\vec{r}, \vec{r}')ds'
\end{gather}$$

$$\begin{gather}
G(\vec{r}, \vec{r}') = \left.\frac{e^{-ik|\vec{r} - \vec{r}'|}}{4\pi |\vec{r} - \vec{r}'|}\right|_{\text{Д.З.}} = e^{ikx'}\cdot \frac{x}{r} e^{iky'\cdot y/r} e^{ikz'\cdot z/r} G(r)
\end{gather}$$

$$\begin{gather}
\begin{cases}
\vec{x}^{0} =  \vec{r}^{0}\sin \theta \cos \alpha + \vec{\theta}^{0}\cos \theta \cos \alpha - \vec{\alpha}^{0}\sin \alpha  \\
\vec{y}^{0} = \vec{r}^{0}\sin \theta \sin\alpha + \vec{\theta}^{0} \cos \theta \sin\alpha + \vec{\alpha}^{0}\cos\alpha  \\
\vec{z}^{0} = \vec{r}^{0}\cos \theta- \vec{\theta}^{0}\sin\theta
\end{cases} \end{gather}$$
$$\begin{gather}
\begin{cases}
\vec{r}^{0} = \vec{x}^{0}\sin \theta \cos\alpha + \vec{y}^{0}\sin \theta \sin\alpha + \vec{z}^{0}\cos\theta \\
\vec{\theta}^{0} = \vec{x}^{0}\cos \theta \cos \alpha + \vec{y}^{0}\cos \theta \sin\alpha - \vec{z}^{0}\sin \theta \\
\vec{\alpha}^{0} = -\vec{x}^{0}\sin\alpha a + \vec{y}^{0}\cos\alpha
\end{cases}
\end{gather}$$

\end{gather}$$
