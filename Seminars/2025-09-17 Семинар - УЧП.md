$$\begin{gather}

$$\begin{gather}

y = \xi-2\eta
\end{cases}, \\

u_{y} = u_{\xi}\xi_{y} + u_{\eta}\eta_{y} = u_{\xi}

$$\begin{gather}
u_{xx} = (u_{x})_{x} = (-2u_{\xi} - u_{\eta})_{x} = \\
= -2(-2u_{\xi \xi} - u_{\xi \eta}) - (-2u_{\eta \xi} - u_{\eta \eta}) = \\
= 4u_{\xi \xi} + 4u_{\xi \eta} + u_{\eta \eta}
\end{gather}$$

$$\begin{gather}
u_{yy} = (u_{y})_{y} = (u_{\xi})_{y} = u_{\xi \xi} \cdot \xi_{y} + u_{\eta \xi}\cdot \eta_{y} = u_{\xi \xi} \\
u_{xy} = \dots = -u_{\xi \xi} - u_{\eta \xi}
\end{gather}$$

$$\begin{gather}
4u_{\xi \xi} + 4u_{\xi \eta} + u_{\eta \eta} + 4(-2u_{\xi \xi} - u_{\eta \xi}) + 5u_{\xi \xi} - (-2u_{\xi} - u_{\eta}) = 2(-\eta) \\
u_{\xi \xi} + u_{\eta \eta} + 2u_{\xi} +u_{\eta} = -2\eta
\end{gather}$$

$$\begin{gather}
u = v\cdot e^{K_{1}\xi + K_{2}\eta} \\
u_{\xi} = v_{\xi}e^{K_{1}\xi + K_{2}\eta} + vK_{1}e^{K_{1}\xi + K_{2}\eta} \\
u_{\eta} = v_{\eta}e^{K_{1}\xi + K_{2}\eta} + vK_{2}e^{K_{1}\xi + K_{2}\eta} \\
u_{\xi \xi} = \text{очев.} \\
u_{\eta \eta} = \text{очев.}
\end{gather}$$

$$\begin{gather}
