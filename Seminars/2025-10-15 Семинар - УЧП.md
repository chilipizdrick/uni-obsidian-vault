### Задача 1

$$\begin{gather}
4u_{xx} + u_{yy} = 0, \ x \in (0,4), y \in(0, \pi) \\
u(0,y) = 8, \ u(4,y) = |y-\frac{\pi}{2}|, \\
u_{y}(x,0) = 0, \ u(x,\pi) = 0 \\
\end{gather}$$
#### I.
$$\begin{gather}
4u_{xx} + u_{yy} = 0 \\
u_{y}(x,0) = 0, \ u(x,\pi) = 0 \\
u = XY \\
\end{gather}$$

$$\begin{gather}
\begin{cases}
4X'' - \mu^{2}X = 0 \\
Y'' + \mu^{2}Y = 0
\end{cases} \\
u_{y}(x,0) = 0 \implies Y'(0) = 0 \\
u(x, \pi) = 0 \implies Y(\pi) = 0 \\
Y = C_{1}\cos(\mu y) + C_{2}\sin(\mu y) \\
Y'(0) = C_{2}\sin(0) = 0 \implies C_{2} = 0 \ \text{или} \ \mu = 0 \\
Y(\pi) = 0 \implies C_{1}\cos \pi \mu = 0 \implies \mu_{n} = \frac{1}{2} + n \\
\end{gather}$$

$$\begin{gather}
Y_{n}(y) = C_{1,n} \cos\left( \left( \frac{1}{2} + n \right)y \right), \ n \in \mathbb{Z} \\
Y_{n}(y) = \cos\left( \left( \frac{1}{2} + n \right)y \right), \ n = 0,1,2,\dots - \text{лнз решения} = \text{базис} \\
\end{gather}$$

$$\begin{gather}
\text{при} \ \mu = 0, Y'' = 0 \implies Y(y) = C_{1}y + C_{2} \\
Y'(0) = C_{1} = 0 \\
Y(\pi) = C_{2} = 0
\end{gather}$$
#### II.
$$\begin{gather}
4X'' - \mu^{2} X = 0, \ 4K^{2} - \mu^{2}_{n} = 0, \ k = \pm \frac{\mu_{n}}{2} \\
X_{n}(x) = A_{n}e^{\mu_{n} x/2} + B_{n}e^{-\mu_{n} x/2}
\end{gather}$$

$$\begin{gather}
u(x,y) = \sum_{n=0}^{\infty} \underbrace{(A_{n}e^{\mu_{n}x/2} + B_{n}e^{-\mu_{n}x/2})}_{X_{n}(x)}\cos\left( \left( \frac{1}{2+n} \right)y \right) \\
\text{или также можно сразу через ряды:} \\
\sum_{n=0}^{\infty} (X_{n}''Y_{n} + X_{n}Y''_{n}) = 0 \\
\end{gather}$$

$$\begin{gather}
y(0,y) = \sum_{n=0}^{\infty} (A_{n} + B_{n})\cos\left( \frac{1}{2} +n \right)y = 8 \\
u(4,y) = \sum_{n=0}^{\infty} (A_{n}e^{\mu_{n}/2} + B_{n}e^{-\mu_{n}/2})\cos\left( \left( \frac{1}{2} + n \right)y \right) = \left|y-\frac{\pi}{2}\right| \\
\end{gather}$$

$$\begin{gather}
f(x) = C_{1}\phi_{1}(x) + C_{2}\phi_{2}(x) + \dots \\
(f, \phi_{1}) = C_{1}(\phi_{1}, \phi_{1}), \ C_{n} = \frac{f,\phi_{n}}{||\phi_{n}||^{2}}
\end{gather}$$

> [!important] Разложение по любому ортогональному базису:
> $$C_{n} = \frac{(f,\phi_{n})}{||\phi_{n}||^{2}}$$

Применим данные знания:

$$\begin{gather}
A_{n} + B_{n} = \frac{f,\phi_{n}}{(\phi_{n},\phi_{n})} = \frac{\int_{0}^{\pi} 8\cdot \cos\left( \frac{1}{2} +n \right)y \, dy}{\int_{0}^{\pi} \cos ^{2}\left( \frac{1}{2} + n \right) y \, dy } \\
\end{gather}$$

$$\begin{gather}
= \frac{\left.-\frac{8}{\frac{1}{2} + n}\sin\left( \frac{1}{2} + n \right)y\right|_{0}^{\pi}}{\frac{1}{2}\int_{0}^{\pi} \left( 1+\cos\left( 1+2n  \right)y \right)  \, dy } = -\frac{\frac{8(-1)^{n}}{\frac{1}{2} + n}}{\frac{\pi}{2}} = \\
= \frac{8(-1)^{n+1}}{\pi(1+2n)}
\end{gather}$$

Первая важная формула получена: $$A_{n} + B_{n} = \frac{8(-1)^{n+1}}{\pi(1+2n)}$$

$$\begin{gather}
(A_{n}e^{2\mu_{n}} + B_{n}e^{-2\mu_{n}}) = \frac{\left( |y-\frac{\pi}{2}|, \cos \left( \frac{1}{2}+n \right)y\right)}{\frac{\pi}{2}} = \\
= \int_{0}^{\pi} \left|y-\frac{\pi}{2}\right|\cos\left( \frac{1}{2} + n \right)y \, dy  = \begin{cases}
\int_{0}^{\pi} \left( y-\frac{\pi}{2} \right)\cos\left( \frac{1}{2} + n \right)y \, dy , y\geq \frac{\pi}{2} \\
-\int_{0}^{\pi} \left( y-\frac{\pi}{2} \right)\cos\left( \frac{1}{2} + n \right)y \, dy , y < \frac{\pi}{2}
\end{cases} \\
\end{gather}$$

$$\begin{gather}
\left.\frac{\frac{\pi}{2}}{\frac{1}{2} + n}\sin\left( \frac{1}{2}  + n\right)y \right|_{0}^{\pi}, \ y \geq \frac{\pi}{2} \\
\begin{cases}
\frac{\frac{\pi}{2}}{\frac{1}{2} + n}(-1)^{n}, \ y \geq \frac{\pi}{2} \\
\frac{\frac{\pi}{2}}{\frac{1}{2} + n}(-1)^{n+1}, \ y < \frac{\pi}{2}
\end{cases}
\end{gather}$$

$$\begin{gather}
\int_{0}^{\pi}y\cos\left( \left( \frac{1}{2} + n \right)y \right) \, dy =\\
= \left.\frac{1}{\frac{1}{2} + n}y\sin\left( \frac{1}{2}+n \right)y \right|_{0}^{\pi} - \frac{1}{\frac{1}{2} + n}\int_{0}^{\pi} \sin\left( \frac{1}{2}+n \right)y \, dy = \\
= \frac{\pi}{\frac{1}{2}+n} (-1)^{n} + \left.\frac{1}{\left(\frac{1}{2} + n\right)^{2}}\cos \left( \frac{1}{2} + n \right)y\right|_{0}^{\pi} = \frac{\pi(-1)^{n}}{\frac{1}{2} + n} - \frac{1}{\left( \frac{1}{2} + n \right)^{2}}
\end{gather}$$

$$\begin{gather}
(A_{n}e^{2\mu_{n}} + B_{n}e^{-2\mu_{n}}) = \begin{cases}
\frac{\pi(-1)^{n}}{1+2n} + \frac{\pi(-1)^{n}}{\frac{1}{2} + n} - \frac{1}{\left( \frac{1}{2} + n \right)^{2}}, \ y \geq \frac{\pi}{2} \\
\frac{\pi(-1)^{n+1}}{1+2n} + \frac{\pi(-1)^{n+1}}{\frac{1}{2} + n} + \frac{1}{\left( \frac{1}{2} + n \right)^{2}}, \ y < \frac{\pi}{2} \\
\end{cases}
\end{gather}$$

### Задача 2

$$\begin{gather}
u_{xx} + 2u_{yy} = xe^{-y}, \ x \in(0,1), \ y \in (0,8) \\
u_{x}(0,y) = 0, \ u_{x}(1,y) = 0 \\
u(x,0) = 0, \ u(x,8) = 0 \\
\end{gather}$$
#### I.
$$\begin{gather}
u_{xx} + 2u_{yy} = 0 \\
u = XY \\
X''Y + 2XY'' = 0 \\
\frac{X''}{X} = -2\frac{Y''}{Y} = \lambda = -\mu^{2} = \mathrm{const} \\
\begin{cases}
X'' + \mu^{2}X = 0 \\
2Y'' -\mu^{2}Y = 0
\end{cases} \\
\end{gather}$$

$$\begin{gather}
X = C_{1}\cos \mu x + C_{2}\sin \mu x \\
X' = -C_{1}\sin \mu x + C_{2}\cos \mu x \\
-C_{1}\cdot 0 + C_{2} \cdot 1 = 0 \implies C_{2} = 0 \\
-C_{1}\sin \mu_{n} = 0 \\
\mu_{n} = \pi n, \ n = 1,2,\dots \\
X_{n} = \cos \pi nx, \ n = 1,2,3,\dots
\end{gather}$$

$$\begin{gather}
\mu = 0 \implies X_{0}(x) = C_{1}x + C_{2} \\
X'(0) = C_{1} = 0 \\
X'(1) = C_{1} = 0 \\
C_{2} = \mathrm{const} \\
X_{0}(x) = C_{2} = 1
\end{gather}$$

$$\begin{gather}
X_{0}(x) = 1 \\
X_{n}(x) = \cos \pi nx, \ n = 1,2,3,\dots
\end{gather}$$

#### II.
$$\begin{gather}
u(x,y) = Y_{0}(y)\cdot 1 \sum_{n=1}^{\infty} Y_{n}(y) \cos(\pi nx) \\
Y_{0}, \ Y_{n} - \text{некоторые неизвестные коэфф.}
\end{gather}$$

$$\begin{gather}
\sum_{n=1}^{\infty} Y_{n}\cdot (-\pi^{2}n^{2})\cos \pi n + 2Y_{0}'' + 2\sum_{n=1}^{\infty} Y''_{n}\cos\pi nx = xe^{-y} = xe^{-y} \\
\end{gather}$$

$$\begin{gather}
2Y''_{0} + \sum_{n=1}^{\infty} (2Y''_{n} - \pi^{2}n^{2}Y_{n})\cos \pi nx = xe^{-y} \\
xe^{-y} = A_{0}(y) + \sum_{n=1}^{\infty} A_{n}(y)\cos \pi nx \\
\end{gather}$$

$$\begin{gather}
A_{0}(y) = 2Y_{0}'' = \frac{xe^{-y}, 1}{(1,1)} = \frac{\frac{e^{-y}}{2}}{1} = \frac{e^{-y}}{2} \implies Y_{0}'' = \frac{e^{-y}}{4} \\
A_{n}(y) = \frac{(xe^{-y}, \cos \pi nx)}{(\cos \pi nx, \cos \pi nx)} = e^{-y}\frac{\int_{0}^{1}x\cos \pi nx \, dx }{\frac{1}{2}\int_{0}^{1} (1+\cos 2\pi nx) \, dx } = \\
2e^{-y} \frac{1}{\pi n}\int_{0}^{1}\sin \pi nx \, dx = 2e^{-y} \cdot \frac{(-1)^{n} - 1}{\pi^{2}n^{2}}
\end{gather}$$

$$\begin{gather}
2Y''_{n} - \pi^{2}n^{2}Y_{n} = e^{-y} \cdot \sigma_{n}, \ \sigma_{n} = \frac{2((-1)^{n} + 1)}{\pi^{2}n^{2}}
\end{gather}$$
