### Задача 1

$$\begin{gather}
4u_{xx} + u_{yy} = 0, \ x \in (0,4), y \in(0, \pi) \\
u(0,y) = 8, \ u(4,y) = |y-\frac{\pi}{2}|, \\
u_{y}(x,0) = 0, \ u(x,\pi) = 0 \\
\end{gather}$$
#### I.
$$\begin{gather}
4u_{xx} + u_{yy} = 0 \\
u_{y}(x,0) = 0, \ u(x,\pi) = 0 \\
u = XY \\
\end{gather}$$

$$\begin{gather}
\begin{cases}
4X'' - \mu^{2}X = 0 \\
Y'' + \mu^{2}Y = 0
\end{cases} \\
u_{y}(x,0) = 0 \implies Y'(0) = 0 \\
u(x, \pi) = 0 \implies Y(\pi) = 0 \\
Y = C_{1}\cos(\mu y) + C_{2}\sin(\mu y) \\
Y'(0) = C_{2}\sin(0) = 0 \implies C_{2} = 0 \ \text{или} \ \mu = 0 \\
Y(\pi) = 0 \implies C_{1}\cos \pi \mu = 0 \implies \mu_{n} = \frac{1}{2} + n \\
\end{gather}$$

$$\begin{gather}
Y_{n}(y) = C_{1,n} \cos\left( \left( \frac{1}{2} + n \right)y \right), \ n \in \mathbb{Z} \\
Y_{n}(y) = \cos\left( \left( \frac{1}{2} + n \right)y \right), \ n = 0,1,2,\dots - \text{лнз решения} = \text{базис} \\
\end{gather}$$

$$\begin{gather}
\text{при} \ \mu = 0, Y'' = 0 \implies Y(y) = C_{1}y + C_{2} \\
Y'(0) = C_{1} = 0 \\
Y(\pi) = C_{2} = 0
\end{gather}$$
#### II.
$$\begin{gather}
4X'' - \mu^{2} X = 0, \ 4K^{2} - \mu^{2}_{n} = 0, \ k = \pm \frac{\mu_{n}}{2} \\
X_{n}(x) = A_{n}e^{\mu_{n} x/2} + B_{n}e^{-\mu_{n} x/2}
\end{gather}$$

$$\begin{gather}
u(x,y) = \sum_{n=0}^{\infty} \underbrace{(A_{n}e^{\mu_{n}x/2} + B_{n}e^{-\mu_{n}x/2})}_{X_{n}(x)}\cos\left( \left( \frac{1}{2+n} \right)y \right) \\
\sum_{n=0}^{\infty} (X_{n}''Y_{n} + X_{n}Y''_{n}) = 0
\end{gather}$$
