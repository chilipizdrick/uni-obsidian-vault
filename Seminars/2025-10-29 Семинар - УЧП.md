### Решение уравнения Пуассона в полярных координатах

### Задача 1

#### I

$$\begin{gather}
\Delta u = 4y \\
\Omega = \{ x,y : \ \sqrt{ x^{2} + y^{2} } < 4 \} \\
u|_{\sigma\Omega} = \cos 4\phi + \sin 2\phi \\
\end{gather}$$

$$\begin{gather}
\Delta u = \frac{1}{r}\cdot (ru_{r})_{r} + \frac{1}{r^{2}}u_{\phi \phi} \\
\end{gather}$$

$$\begin{gather}
\begin{cases}
\frac{1}{r}(r\cdot u_{r})_{r} + \frac{1}{r^{2}}u_{\phi \phi} = 4r \sin \phi, \ \Omega: r<4 \\
u(4,\phi) = \cos 4\phi + \sin 2\phi \\
u(r, \phi + 2\pi) = u(r, \phi) - \text{однозн.}
\end{cases}
\end{gather}$$

$$\begin{gather}
\begin{cases}
\frac{1}{r}(ru_{r})_{r} + \frac{1}{r^{2}}u_{\phi \phi} = 0 \\
u(r,\phi+2\pi) = u(r,p\phi)
\end{cases} \\
u = R(r) \cdot \Phi(\phi) \\
\end{gather}$$

$$\begin{gather}
\frac{1}{r}(R (R'\Phi))_{r} + \frac{1}{r^{2}}R\Phi'' = 0 \\
\frac{1}{r}(R'^{2}\Phi + RR''\Phi) + \frac{1}{r^{2}}R\Phi'' = 0 | \cdot \frac{r^{2}}{R\Phi} \\
r\left( \frac{R'^{2}}{R} + R'' \right) = -\frac{\Phi''}{\Phi} = \lambda = -\mu^{2} \\
\end{gather}$$

$$\begin{gather}
\begin{cases}
\Phi'' + \mu^{2}\Phi = 0 \\
\Phi(\phi + 2\pi) = \Phi(\phi)
\end{cases} \\
\Phi = C_{1}\cos \mu\phi + C_{2}\sin \mu\phi \\
C_{1}\cos (\mu\phi + 2\mu\pi) + C_{2} \sin (\mu\phi + 2\mu\pi) = C_{1}\cos(\mu \phi) + C_{2}\sin(\mu \phi) \\
\mu_{n} = n, \ n \in \mathbb{Z} \\
\end{gather}$$

$$\begin{gather}
\Phi_{n}(\phi) = \left[\begin{array}{l}
1 \\
\cos n\phi, \ n \in \mathbb{N} \\
\sin n\phi, \ n \in \mathbb{N}
\end{array}\right.
\end{gather}$$

#### I

$$\begin{gather}
u(r,\phi) = 1\cdot a_{0}(r) + \sum_{n=1}^{\infty} (a_{n}(r) \cos n\phi + b_{n}(r) \sin n\phi)
\end{gather}$$

$$\begin{gather}
r(ra_{0}') + \sum_{n=1}^{\infty}  (r(ra_{n}')' - n^{2} a_{n})\cos n\phi + \\
+ \sum_{n=1}^{\infty} (r(rb_{n}')' - n^{2}b_{n})\sin n\phi = 4r^{3}\sin \phi
\end{gather}$$

Откуда получаем:
$$\begin{gather}
\begin{cases}
r(ra_{0}')' = 0 \\
r(ra_{n}')' - n^{2}a_{n} = 0 \\
r(rb_{1}')' - b_{1} = 4r^{3} \\
r(rb_{n}')' - n^{2}b_{n} = 0
\end{cases} \\
\end{gather}$$

$$\begin{gather}
a_{0} = A_{0}\ln r + B_{0} \\
\end{gather}$$

$$\begin{gather}
r(a'_{n} + ra''_{n}) - n^{2}a_{n} = 0 \\
r^{2}a'' + ra_{n}' - n^{2}a_{n} = 0 \\
a_{n} = r^{\alpha} \\
\alpha(\alpha-1) + \alpha-n^{2} = 0 \implies \alpha = \pm n \\
a_{n} = A_{n}r^{n} + \frac{B_{n}}{r^{n}} \\
\end{gather}$$

$$\begin{gather}
b_{n} = \tilde{A}_{n} r^{n} + \frac{\tilde{B}_{n}}{r^{n}} \\
\end{gather}$$

$$\begin{gather}
b_{1}^{оо} = \tilde{A}_{1}r + \frac{\tilde{B}_{1}}{r} \\
b_{1}^{чн} = Cr^{3} \\
Cr^{2}\cdot 6r^{1} + Cr\cdot 3r^{2} - C r^{3} = 4r^{3} \\
C = \frac{1}{2} \\
b_{1} = \tilde{A}_{1}r + \frac{\tilde{B}_{1}}{r} + \frac{1}{2}r^{3}
\end{gather}$$

В силу условия ограниченности:
$$\begin{gather}
A_{0} = 0, \ B_{n} = 0, \ \tilde{B}_{n} = 0 \\
u(r,\phi) = B_{0}+ \sum_{n=1}^{\infty} A_{n}r^{n}\cos n\phi + \\
+ \left(\tilde{A}_{1}r + \frac{1}{2}r^{3}\right)\sin \phi + \sum_{n=2}^{\infty} \tilde{A}_{n}r^{n} \sin n\phi
\end{gather}$$

$$\begin{gather}
u(4,\phi) = B_{0} + \sum_{n=1}^{\infty} A_{n}4^{n} \sin n\phi + (\tilde{A}_{1}\cdot 4 + 32)\sin \phi + \\
+ \sum_{i=2}^{n} \tilde{A}_{n}4^{n}\sin n\phi = \cos 4\phi + \sin 2\phi
\end{gather}$$

$$\begin{gather}
B_{0} = 0 \\
A_{n}\cdot 4^{n} = 0, \ n \neq 4 \\
A_{4}\cdot 4^{4} = 1 \implies A_{4} = \frac{1}{256} \\
\end{gather}$$

$$\begin{gather}
\tilde{A}_{1} = -8 \\
\tilde{A}_{n} \cdot 4^{4} = 0 \\
\tilde{A}_{n} = 0 \\
\tilde{A}_{2} = \frac{1}{16}
\end{gather}$$

$$\begin{gather}
\frac{1}{256} r^{4}\cos 4\phi + \left( -8r + \frac{1}{2}r^{3} \right)\sin \phi + \frac{1}{16}
\end{gather}$$
