### Решение уравнения Пуассона в полярных координатах

### Задача 1

#### I

$$\begin{gather}
\Delta u = 4y \\
\Omega = \{ x,y : \ \sqrt{ x^{2} + y^{2} } < 4 \} \\
u|_{\sigma\Omega} = \cos 4\phi + \sin 2\phi \\
\end{gather}$$

$$\begin{gather}
\Delta u = \frac{1}{r}\cdot (ru_{r})_{r} + \frac{1}{r^{2}}u_{\phi \phi} \\
\end{gather}$$

$$\begin{gather}
\begin{cases}
\frac{1}{r}(r\cdot u_{r})_{r} + \frac{1}{r^{2}}u_{\phi \phi} = 4r \sin \phi, \ \Omega: r<4 \\
u(4,\phi) = \cos 4\phi + \sin 2\phi \\
u(r, \phi + 2\pi) = u(r, \phi) - \text{однозн.}
\end{cases}
\end{gather}$$

$$\begin{gather}
\begin{cases}
\frac{1}{r}(ru_{r})_{r} + \frac{1}{r^{2}}u_{\phi \phi} = 0 \\
u(r,\phi+2\pi) = u(r,p\phi)
\end{cases} \\
u = R(r) \cdot \Phi(\phi) \\
\end{gather}$$

$$\begin{gather}
\frac{1}{r}(R (R'\Phi))_{r} + \frac{1}{r^{2}}R\Phi'' = 0 \\
\frac{1}{r}(R'^{2}\Phi + RR''\Phi) + \frac{1}{r^{2}}R\Phi'' = 0 | \cdot \frac{r^{2}}{R\Phi} \\
r\left( \frac{R'^{2}}{R} + R'' \right) = -\frac{\Phi''}{\Phi} = \lambda = -\mu^{2} \\
\end{gather}$$

$$\begin{gather}
\begin{cases}
\Phi'' + \mu^{2}\Phi = 0 \\
\Phi(\phi + 2\pi) = \Phi(\phi)
\end{cases} \\
\Phi = C_{1}\cos \mu\phi + C_{2}\sin \mu\phi \\
C_{1}\cos (\mu\phi + 2\mu\pi) + C_{2} \sin (\mu\phi + 2\mu\pi) = C_{1}\cos(\mu \phi) + C_{2}\sin(\mu \phi) \\
\mu_{n} = n, \ n \in \mathbb{Z} \\
\end{gather}$$

$$\begin{gather}
\Phi_{n}(\phi) = \left[\begin{array}{l}
1 \\
\cos n\phi, \ n \in \mathbb{N} \\
\sin n\phi, \ n \in \mathbb{N}
\end{array}\right.
\end{gather}$$

$$\begin{gather}
u(r,\phi) = 1\cdot a_{0}(r) + \sum_{n=1}^{\infty} (a_{n}(r) \cos n\phi + b_{n}(r) \sin n\phi)
\end{gather}$$

$$\begin{gather}
r(ra_{0}') + \sum_{n=1}^{\infty}  (r(ra_{n}')' - n^{2} a_{n})\cos n\phi + \\
+ \sum_{n=1}^{\infty} (r(rb_{n}')' - n^{2}b_{n})\sin n\phi = 4r^{3}\sin \phi

$$\begin{gather}
