### Задача 2

![[2025-10-14 Семинар - Механика 2025-10-14 11.17.14.excalidraw]]

$$\begin{gather}
v_{a} > v_{b} \\
R - ! \\
\hline
v_{C} - ?
\end{gather}$$

$$\begin{gather}
\begin{cases}
v_{a} = v_{A} \\
v_{b} = b_{B} 
\end{cases} - ! \\
x = CP \\
\omega = \frac{v_{A}}{R+x} = \frac{v_{B}}{R-x} \\
v_{A} (R-x) =v_{B}(R+x) \\
x = \frac{R(v_{A} - v_{B})}{v_{B} + v_{A}} \\
\end{gather}$$

$$\begin{gather}
\omega = \frac{v_{A}}{R + \frac{R(v_{A} - v_{B})}{v_{B} + v_{A}}} \\
\omega = \frac{v_{A}(v_{B} + v_{A})}{2Rv_{A}} \\
\omega = \frac{v_{B}+v_{A}}{2R} - \text{победная скорость, надо запомнить} \\
v_{C}  = \frac{v_{A} - v_{B}}{2}
\end{gather}$$

### Задание 3

![[2025-10-14 Семинар - Механика 2025-10-14 11.38.58.excalidraw]]

$$\begin{gather}
OA = r \\
\omega = \mathrm{const} \\
\phi - ! \\
\hline
v_{B} - ?
\end{gather}$$

$$\begin{gather}
v_{A} = \omega r \\
\vec{v}_{A} \perp OA \\
\angle OAD = \phi + \beta \\
\alpha = \frac{\pi}{2} - (\phi + \beta) \\
v_{A} = \cos \alpha = v_{B} \cdot \cos\beta \\
v_{B} = \omega r \cdot \frac{\cos\alpha}{\cos\beta} \\
v_{B} = \omega r\cdot \frac{\sin (\phi + \beta)}{\cos\beta}
\end{gather}$$

$$\begin{gather}
v_{B} = \frac{\omega r}{\cos\beta} \left[ \sin \phi \cos\beta + \sin \beta \cos \phi \right] \\
v_{B} = \omega r \left[ \sin \phi + \cos\phi \tan\beta \right] 
\end{gather}$$

$$\begin{gather}
v_{B} = \omega r \left[ \sin \phi + \frac{\cos \phi \cdot \sin\beta}{\sqrt{ 1-\sin ^{2}\beta }} \right]  \\
\frac{\sin \phi}{l} = \frac{\sin \beta}{r} \\
\sin\beta = \frac{r}{l} \sin \phi \\
\end{gather}$$

$$\begin{gather}
v_{B} = \omega r \left[ \sin \phi + \frac{\cos \phi \cdot r \cdot \sin \phi}{l \sqrt{ 1 - \left(\frac{r}{l}\right)^{2} \sin ^{2} \phi }} \right] \\
v_{B} = \omega r \left[ \sin \phi + \frac{r \sin 2\phi}{2l \sqrt{ 1 - \left(\frac{r}{l}\right)^{2} \sin ^{2} \phi }} \right] \\
\end{gather}$$
