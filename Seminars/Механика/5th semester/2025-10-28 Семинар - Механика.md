$$\begin{gather}
v, a, \\
\rho,r,R \\
\hline
\omega_{2}, \omega_{3} - ? \\

v_{A}, v_{B}, v_{C}, v_{D} - ? \\
a_{A}, a_{B}, a_{C}, a_{D} - ? \\
\end{gather}$$

Система состоит из 3-ех тел, которые движутся следующим образом:
1. Тело 1 движется поступательно
2. Тело 2 движется вращательно
3. Тело 3 движется плоско-параллельно

$D$ - мгновенный центр скоростей.
$v_{D} = 0$

$$\begin{gather}
v_{2} = v_{A} = v \\
v_{B} = v_{A} = v \\
\omega_{2} = \frac{v_{A}}{\rho} = \frac{v}{\rho}
\end{gather}$$

$$\begin{gather}
\omega_{3} = \frac{v_{F}}{R-r} = \frac{v}{R-r} \\
\frac{v}{R-r} = \frac{v_{C}}{r} \\
v_{C} = \frac{vr}{R-r}
\end{gather}$$

$$\begin{gather}
\vec{a}_{A} = \vec{a}_{A}^{n} + \vec{a}_{A} ^{\tau} \\
a_{A} = \sqrt{ (a_{A}^{n})^{2} + (a_{A}^{\tau})^{2} } \\
a_{A}^{n} = \omega_{2}^{2} \cdot \rho = \frac{v^{2}}{\rho^{2}} \cdot \rho = \frac{v^{2}}{\rho} \\
\epsilon_{2} = \dot{\omega}_{2} = \frac{\dot{v}}{\rho} = \frac{a}{\rho} \\
a_{A}^{\tau} = \epsilon_{2}\cdot \rho = a \\
a_{A} = \sqrt{ a^{2} + \frac{v^{4}}{\rho^{2}} }
\end{gather}$$

$$\begin{gather}
a_{B} = a^{\tau}_{A} = a \\
\vec{a}_{F} = \vec{a}^{\tau}_{F} + \vec{a}^{n}_{F} = \vec{a}_{B} + \vec{a}^{n}_{F} \\
a_{F} = \sqrt{ (a^{\tau}_{A})^{2} + a^{n}_{FD} } \\
a^{n}_{FD} = \omega^{2}_{3} \cdot (R-r) = \frac{v^{2}}{R-r} \\
a_{F} = \sqrt{ a^{2} + \frac{v^{4}}{(R-r)^{2}} } \\
\epsilon_{3} = \dot{\omega}_{3} = \frac{\dot{v}}{R-r} = \frac{a}{R-r}
\end{gather}$$

$$\begin{gather}
\vec{a}_{C} = \vec{a}_{C}^{\tau} + \vec{a}_{C}^{n} \\
a_{C} = \epsilon_{3}\cdot r = \frac{ar}{R-r} \\
\end{gather}$$

$C$ - полюс.

$$\begin{gather}
\vec{a}_{D} = \vec{a}_{C} + \vec{a}_{DC}^{n} + \vec{a}_{DC}^{\tau} \\
a^{n}_{DC} = \omega_{3}^{2}\cdot r = \frac{v^{2}r}{(R-r)^{2}} \\
a^{\tau}_{DC} = \epsilon_{3} \cdot r = \frac{ar}{R-r}
\end{gather}$$

$$\begin{gather}
a_{D} = \sqrt{ a_{Dx}^{2} + a_{Dy}^{2} } \\
a_{Dx} = a_{C} - a_{DC}^{\tau} = \frac{ar}{R-r} - \frac{ar}{R-r} = 0 \\
a_{Dy} = a^{n}_{DC} = \omega_{3}^{2}\cdot r = \frac{v^{2}r}{(R-r)^{2}} \\
a_{D} = \frac{v^{2}r}{(R-r)^{2}} \\
\vec{}
\end{gather}$$
